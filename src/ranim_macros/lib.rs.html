14fn ranim_path() -> proc_macro2::TokenStream {
15    match crate_name(RANIM_CRATE_NAME) {
16        Ok(FoundCrate::Name(name)) => {
17            let ident = Ident::new(&name, Span::call_site());
18            quote!(::#ident)
19        }
20        // The canonical self import in ranim also works in doctests, whose
21        // CARGO_CRATE_NAME still names the library rather than the test crate.
22        _ => quote!(::ranim),
23    }
24}

26fn ranim_core_path() -> proc_macro2::TokenStream {
27    if let Ok(res) = crate_name("ranim-core") {
28        match (res, std::env::var("CARGO_CRATE_NAME").as_deref()) {
29            (FoundCrate::Itself, Ok("ranim-core") | Ok("ranim_core")) => return quote!(crate),
30            (FoundCrate::Name(name), _) => {
31                let ident = Ident::new(&name, Span::call_site());
32                return quote!(::#ident);
33            }
34            _ => (),
35        }
36    } else if let Ok(res) = crate_name("ranim") {
37        match (res, std::env::var("CARGO_CRATE_NAME").as_deref()) {
38            (FoundCrate::Itself, Ok("ranim")) => return quote!(crate::core),
39            (FoundCrate::Name(name), _) => {
40                let ident = Ident::new(&name, Span::call_site());
41                return quote!(::#ident::core);
42            }
43            _ => (),
44        }
45    }
46    ranim_path()
47}

49/// 解析单个属性（#[scene(...)] /  / #[output(...)]）
50#[derive(Default)]
51struct SceneAttrs {
52    name: Option<String>,        // #[scene(name = "...")]
53    clear_color: Option<String>, // #[scene(clear_color = "#000000")]
54    wasm_demo_doc: bool,         // #[wasm_demo_doc]

71// MARK: scene
72#[proc_macro_attribute]
73pub fn scene(args: TokenStream, input: TokenStream) -> TokenStream {
74    let ranim = ranim_path();
75    let input_fn = parse_macro_input!(input as ItemFn);
76    let attrs = parse_scene_attrs(args, input_fn.attrs.as_slice()).unwrap();
77
78    let fn_name = &input_fn.sig.ident;
79    // `#[wasm_demo_doc]` scenes are embedded into rustdoc pages, so make sure
80    // they are public enough for rustdoc to render them.
81    let vis = if attrs.wasm_demo_doc {
82        quote!(pub)
83    } else {
84        let vis = &input_fn.vis;
85        quote!(#vis)
86    };
87    let signature = &input_fn.sig;
88    let fn_body = &input_fn.block;
89    let doc_attrs: Vec<_> = input_fn
90        .attrs
91        .iter()
92        .filter(|attr| attr.path().is_ident("doc"))
93        .collect();
94
95    // 场景名称
96    let scene_name = attrs.name.unwrap_or_else(|| fn_name.to_string());
97
98    // Scene configuration
99    let clear_color = attrs.clear_color.unwrap_or("#333333ff".to_string());
100    let scene_config = quote! {
101        #ranim::SceneConfig {
102            clear_color: #clear_color.to_string(),
103        }
104    };
105
106    // Owned output descriptors
107    let mut outputs = Vec::new();
108    for OutputDef {
109        width,
110        height,
111        fps,
112        save_frames,
113        name,
114        name_template,
115        dir,
116        format,
117    } in attrs.outputs
118    {
119        let name_token = match name.as_deref() {
120            Some(n) if !n.is_empty() => quote! { Some(#n.to_string()) },
121            _ => quote! { None },
122        };
123        let name_template_token = match name_template.as_deref() {
124            Some(n) if !n.is_empty() => quote! { Some(#n.to_string()) },
125            _ => quote! { None },
126        };
127        let format_token = match format.as_deref() {
128            Some("mp4") | None => quote! { #ranim::OutputFormat::Mp4 },
129            Some("webm") => quote! { #ranim::OutputFormat::Webm },
130            Some("mov") => quote! { #ranim::OutputFormat::Mov },
131            Some("gif") => quote! { #ranim::OutputFormat::Gif },
132            Some(other) => panic!("unknown output format: {other:?}"),
133        };
134        outputs.push(quote! {
135            #ranim::Output {
136                width: #width,
137                height: #height,
138                fps: #fps,
139                save_frames: #save_frames,
140                name: #name_token,
141                name_template: #name_template_token,
142                dir: #dir.to_string(),
143                format: #format_token,
144            }
145        });
146    }
147    if outputs.is_empty() {
148        outputs.push(quote! {
149            #ranim::Output::default()
150        });
151    }
152
153    let doc = if attrs.wasm_demo_doc {
154        quote! {
155            #[doc = concat!("<div class=\"ranim-preview\" style=\"display:grid;gap:8px\"><ranim-viewport data-module-path=\"", module_path!(), "\" scene=\"", #scene_name, "\"></ranim-viewport><ranim-controls></ranim-controls></div>\n\n")]
156        }
157    } else {
158        quote! {}
159    };
160
161    let scene = quote! {
162        let mut scene = #ranim::Scene::new(#scene_name, super::#fn_name);
163        scene.config = #scene_config;
164        scene.outputs = vec![#(#outputs),*];
165        scene
166    };
167
168    // ANCHOR: SCENE_MACRO
169    let expanded = quote! {
170        #doc
171        #(#doc_attrs)*
172        #vis #signature #fn_body
173
174        #[doc(hidden)]
175        #vis mod #fn_name {
176            /// Construct this scene's descriptor for explicit registration in an App.
177            pub fn scene() -> #ranim::Scene {
178                #scene
179            }
180        }
181    };
182    // ANCHOR_END: SCENE_MACRO
183
184    TokenStream::from(expanded)
185}

215#[proc_macro_derive(Fill)]
216pub fn derive_fill(input: TokenStream) -> TokenStream {
217    let core = ranim_core_path();
218    impl_derive(input, quote! {#core::traits::Fill}, |field_positions| {
219        quote! {
220            fn set_fill_opacity(&mut self, opacity: f32) -> &mut Self {
221                #(
222                    self.#field_positions.set_fill_opacity(opacity);
223                )*
224                self
225            }

227                [#(self.#field_positions.fill_color(), )*].first().cloned().unwrap()

230                #(
231                    self.#field_positions.set_fill_color(color);

234            }
235        }

292#[proc_macro_derive(Empty)]
293pub fn derive_empty(input: TokenStream) -> TokenStream {
294    let core = ranim_core_path();
295    let input = parse_macro_input!(input as DeriveInput);
296    let name = &input.ident;
297    let generics = &input.generics;
298    let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
299
300    let fields = match &input.data {
301        Data::Struct(data) => &data.fields,
302        _ => panic!("Empty can only be derived for structs"),
303    };
304
305    let field_impls = match fields {
306        Fields::Named(fields) => {
307            let (field_names, field_types): (Vec<_>, Vec<_>) =
308                fields.named.iter().map(|f| (&f.ident, &f.ty)).unzip();
309
310            quote! {
311                Self {
312                    #(
313                        #field_names: #field_types::empty(),
314                    )*
315                }
316            }
317        }
318        Fields::Unnamed(fields) => {
319            let field_types = fields.unnamed.iter().map(|f| &f.ty);
320            quote! {
321                Self (
322                    #(
323                        #field_types::empty(),
324                    )*
325                )
326            }
327        }
328        Fields::Unit => quote! {},
329    };
330
331    let expanded = quote! {
332        impl #impl_generics #core::traits::Empty for #name #ty_generics #where_clause {
333            fn empty() -> Self {
334                #field_impls
335            }
336        }
337    };
338
339    TokenStream::from(expanded)
340}

468fn impl_derive(
469    input: TokenStream,
470    trait_path: proc_macro2::TokenStream,
471    impl_token: impl Fn(Vec<proc_macro2::TokenStream>) -> proc_macro2::TokenStream,
472) -> TokenStream {
473    let input = parse_macro_input!(input as DeriveInput);
474    let name = &input.ident;
475    let generics = &input.generics;
476    let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
477
478    let fields = match &input.data {
479        Data::Struct(data) => &data.fields,
480        _ => panic!("Can only be derived for structs"),
481    };
482
483    let field_positions = get_field_positions(fields)
484        .ok_or("cannot get field from unit struct")
485        .unwrap();
486
487    let impl_token = impl_token(field_positions);
488    let expanded = quote! {
489        impl #impl_generics #trait_path for #name #ty_generics #where_clause {
490            #impl_token
491        }
492    };
493
494    TokenStream::from(expanded)
495}
496
497fn get_field_positions(fields: &Fields) -> Option<Vec<proc_macro2::TokenStream>> {
498    match fields {
499        Fields::Named(fields) => Some(
500            fields
501                .named
502                .iter()
503                .map(|f| {
504                    let pos = &f.ident;
505                    quote! { #pos }
506                })
507                .collect::<Vec<_>>(),
508        ),
509        Fields::Unnamed(fields) => Some(
510            (0..fields.unnamed.len())
511                .map(syn::Index::from)
512                .map(|i| {
513                    quote! { #i }
514                })
515                .collect::<Vec<_>>(),
516        ),
517        Fields::Unit => None,
518    }
