1//! Ranim is an animation engine written in rust based on wgpu, inspired by [3b1b/manim](https://github.com/3b1b/manim/) and [jkjkil4/JAnim](https://github.com/jkjkil4/JAnim).
2//!
3//! A project binary registers its scenes and hands control to the command
4//! line interface:
5//!
6//! ```ignore
7//! use ranim::prelude::*;
8//!
9//! #[scene]
10//! fn hello(r: &mut RanimScene) { /* ... */ }
11//!
12//! fn main() {
13//!     // cargo run -- preview | output | render hello | inspect scenes
14//!     let mut app = App::new();
15//!     app.add_scenes([hello::scene()]);
16//!     run_cli(app.build());
17//! }
18//! ```
19//!
20//! Plugins organize larger projects and custom render configurations:
21//!
22//! ```
23//! use ranim::{App, Plugin, Resource};
24//!
25//! /// Project resources must implement bevy's `Resource`.
26//! #[derive(Resource)]
27//! struct ProjectConfig(u32);
28//!
29//! struct ProjectPlugin;
30//!
31//! impl Plugin for ProjectPlugin {
32//!     fn build(&self, app: &mut bevy_app::App) {
33//!         app.insert_resource(ProjectConfig(42));
34//!     }
35//! }
36//!
37//! let mut app = App::new();
38//! app.add_plugins(ProjectPlugin);
39//! assert_eq!(app.world().resource::<ProjectConfig>().0, 42);
40//! ```
41//!
42//! ## Coordinate System
43//!
44//! Ranim's coordinate system is right-handed coordinate:
45//!
46//! ```text
47//!      +Y
48//!      |
49//!      |
50//!      +----- +X
51//!    /
52//! +Z
53//! ```
54//!
