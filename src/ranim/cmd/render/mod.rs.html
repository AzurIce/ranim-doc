23/// Render the output basename template by replacing placeholders.
24///
25/// Supported placeholders:
26/// - `{name}`: the scene/output name
27/// - `{width}`: output width in pixels
28/// - `{height}`: output height in pixels
29/// - `{fps}`: output frame rate
30fn render_output_basename(template: &str, name: &str, width: u32, height: u32, fps: u32) -> String {
31    template
32        .replace("{name}", name)
33        .replace("{width}", &width.to_string())
34        .replace("{height}", &height.to_string())
35        .replace("{fps}", &fps.to_string())
36}

38/// Render a scene with all its outputs
39pub fn render_scene(scene: &Scene, buffer_count: usize) {
40    for (i, output) in scene.outputs.iter().enumerate() {
41        info!(
42            "Rendering output {}/{} ({})",
43            i + 1,
44            scene.outputs.len(),
45            output.format
46        );
47        render_scene_output(
48            scene.constructor,
49            scene.name.to_string(),
50            &scene.config,
51            output,
52            buffer_count,
53        );
54    }
55}

68/// A single rendering job: one [`Output`] plus whether capture marks should
69/// be processed after the main pass.
70///
71/// The CLI currently only creates full-scene jobs. Time-range and still-frame
72/// rendering will extend this type together with the debug rendering commands.
73pub struct RenderJob {
74    /// Output settings used for this job.
75    pub output: Output,
76    /// Render and save every [`TimeMark::Capture`] after the main pass.
77    ///
78    /// This is enabled by `ranim output` (declared outputs) and disabled by
79    /// the ad-hoc `ranim render` command.
80    pub capture_marks: bool,
81}

84    /// Creates a job that renders the whole scene.
85    pub fn full(output: Output, capture_marks: bool) -> Self {
86        Self {
87            output,
88            capture_marks,
89        }
90    }

136/// Build the scene and run a [`RenderJob`].
137pub fn render_scene_job(
138    constructor: impl SceneConstructor,
139    name: String,
140    scene_config: &SceneConfig,
141    job: RenderJob,
142    buffer_count: usize,
143    on_progress: Option<Box<dyn Fn(u64, u64) + Send>>,
144) {
145    use std::time::Instant;
146
147    let RenderJob {
148        output,
149        capture_marks,
150    } = job;
151
152    info!(
153        "Output: {}x{} {}fps {} dir={:?} save_frames={}",
154        output.width, output.height, output.fps, output.format, output.dir, output.save_frames
155    );
156
157    let t = Instant::now();
158    let scene = constructor.build_scene();
159    trace!("Build timeline cost: {:?}", t.elapsed());
160
161    // Default logic grid 120 Hz (per the time model design); render fps only
162    // decides which logic states are sampled.
163    let mut evaluator = scene.into_evaluator(DEFAULT_LOGIC_FPS);
164
165    let mut app = RanimRenderApp::new(name, scene_config, &output, buffer_count);
166    app.render_scene_with_progress(&mut evaluator, on_progress);
167
168    if capture_marks && !evaluator.time_marks().is_empty() {
169        app.render_capture_marks(&mut evaluator);
170    }
171}

173/// Default logic grid resolution (Hz), per the time model design.
174const DEFAULT_LOGIC_FPS: f64 = 120.0;

176/// Handle to the background render thread, used to submit frame data from the main thread.
177///
178/// Dropping this handle closes the submission channel and terminates the worker thread loop.
179pub struct RenderThreadHandle {
180    submit_frame_tx: async_channel::Sender<RenderFrame>,
181    back_rx: async_channel::Receiver<RenderFrame>,
182    worker_rx: async_channel::Receiver<RenderWorker>,
183}
184
185impl RenderThreadHandle {
186    /// Reuses a [`RenderFrame`], applies `f` to it, and submits it to the background render thread.
187    pub fn sync_and_submit(&self, f: impl FnOnce(&mut RenderFrame)) {
188        let mut store = self.get_store();
189        f(&mut store);
190        self.submit_frame_tx.send_blocking(store).unwrap();
191    }
192
193    /// Retrieves a previously rendered [`RenderFrame`] from the background thread for reuse.
194    pub fn get_store(&self) -> RenderFrame {
195        self.back_rx.recv_blocking().unwrap()
196    }
197
198    /// Closes the submission channel, waits for the background thread to finish, and returns the [`RenderWorker`].
199    pub fn retrive(&self) -> RenderWorker {
200        self.submit_frame_tx.close(); // This terminates the worker thread loop
201        self.worker_rx.recv_blocking().unwrap()
202    }
203}

205/// Background rendering worker that owns the wgpu context, renderer, output textures, and video/image writers.
206pub struct RenderWorker {
207    ctx: WgpuContext,
208    renderer: Renderer,
209    render_textures: Vec<RenderTextures>,
210    clear_color: wgpu::Color,
211    // video writer
212    video_writer: Option<FileWriter>,
213    video_writer_builder: Option<FileWriterBuilder>,
214    save_frames: bool,
215    output_dir: PathBuf,
216    scene_name: String,
217    width: u32,
218    height: u32,
219    fps: u32,
220}

222impl RenderWorker {
223    /// Creates a new [`RenderWorker`].
224    ///
225    /// This checks for or downloads `ffmpeg`, initializes the wgpu context and renderer, and configures the video writer from [`Output`].
226    ///
227    /// # Panics
228    ///
229    /// Panics if `buffer_count < 1`.
230    pub fn new(
231        scene_name: String,
232        scene_config: &SceneConfig,
233        output: &Output,
234        buffer_count: usize,
235    ) -> Self {
236        assert!(buffer_count >= 1, "buffer_count must be at least 1");
237        info!("Checking ffmpeg...");
238        let t = Instant::now();
239        if let Ok(ffmpeg_path) = which::which("ffmpeg") {
240            info!("ffmpeg found at {ffmpeg_path:?}");
241        } else {
242            use std::path::Path;
243
244            info!(
245                "ffmpeg not found from path env, searching in {:?}...",
246                Path::new("./").canonicalize().unwrap()
247            );
248            if Path::new("./ffmpeg").exists() {
249                info!("ffmpeg found at current working directory")
250            } else {
251                info!("ffmpeg not found at current working directory, downloading...");
252                download_ffmpeg("./").expect("failed to download ffmpeg");
253            }
254        }
255        trace!("Check ffmmpeg cost: {:?}", t.elapsed());
256
257        let t = Instant::now();
258        info!("Creating wgpu context...");
259        let ctx = pollster::block_on(WgpuContext::new());
260        trace!("Create wgpu context cost: {:?}", t.elapsed());
261
262        let mut output_dir = PathBuf::from(&output.dir);
263        if !output_dir.is_absolute() {
264            output_dir = std::env::current_dir().unwrap().join(output_dir);
265        }
266        let renderer = Renderer::new(&ctx, output.width, output.height, 8);
267        let render_textures: Vec<RenderTextures> = (0..buffer_count)
268            .map(|_| renderer.new_render_textures(&ctx))
269            .collect();
270        let clear_color = color::try_color(&scene_config.clear_color)
271            .unwrap_or(color::color("#333333ff"))
272            .convert::<LinearSrgb>();
273        let [r, g, b, a] = clear_color.components.map(|x| x as f64);
274        let clear_color = wgpu::Color { r, g, b, a };
275        let (_, _, ext) = output.format.encoding_params();
276        Self {
277            ctx,
278            renderer,
279            render_textures,
280            clear_color,
281            video_writer: None,
282            video_writer_builder: Some(
283                FileWriterBuilder::default()
284                    .with_fps(output.fps)
285                    .with_size(output.width, output.height)
286                    .with_file_path(output_dir.join({
287                        let template = output
288                            .name_template
289                            .as_deref()
290                            .unwrap_or("{name}_{width}x{height}_{fps}");
291                        let base_name = render_output_basename(
292                            template,
293                            output.name.as_deref().unwrap_or(&scene_name),
294                            output.width,
295                            output.height,
296                            output.fps,
297                        );
298                        format!("{base_name}.{ext}")
299                    }))
300                    .with_output_format(output.format),
301            ),
302            save_frames: output.save_frames,
303            output_dir,
304            scene_name,
305            width: output.width,
306            height: output.height,
307            fps: output.fps,
308        }
309    }

311    /// Returns the directory path used to save individual frame images.
312    pub fn save_frame_dir(&self) -> PathBuf {
313        self.output_dir.join(format!(
314            "{}_{}x{}_{}-frames",
315            self.scene_name, self.width, self.height, self.fps
316        ))
317    }

319    /// Moves this [`RenderWorker`] into a newly spawned background thread and returns a [`RenderThreadHandle`].
320    ///
321    /// The background thread loops over incoming [`RenderFrame`]s, rendering and outputting frames using multi-buffered async readback.
322    pub fn yeet(self) -> RenderThreadHandle {
323        let (submit_frame_tx, submit_frame_rx) = async_channel::bounded(1);
324        let (back_tx, back_rx) = async_channel::bounded(1);
325        let (worker_tx, worker_rx) = async_channel::bounded(1);
326
327        back_tx.send_blocking(RenderFrame::default()).unwrap();
328        std::thread::spawn(move || {
329            let mut worker = self;
330            let n = worker.render_textures.len();
331            let mut frame_count = 0u64;
332            let mut cur = 0usize;
333            let mut pending: VecDeque<(usize, u64)> = VecDeque::new();
334
335            while let Ok(store) = submit_frame_rx.recv_blocking() {
336                // Drain oldest pending readback if all targets are occupied
337                if pending.len() >= n {
338                    let (prev, prev_fc) = pending.pop_front().unwrap();
339                    worker.render_textures[prev].finish_readback(&worker.ctx);
340                    worker.output_frame_from(prev, prev_fc);
341                }
342
343                // Render current frame and start async readback
344                worker.renderer.render_frame(
345                    &mut worker.render_textures[cur],
346                    worker.clear_color,
347                    &store,
348                );
349                worker.render_textures[cur].start_readback(&worker.ctx);
350
351                pending.push_back((cur, frame_count));
352                frame_count += 1;
353                cur = (cur + 1) % n;
354
355                // Return store early so main thread can eval next frame
356                // while GPU processes the readback
357                back_tx.send_blocking(store).unwrap();
358
359                // Now try to drain any completed readbacks while we wait
360                // for the next frame from the main thread
361                while let Some(&(prev, _)) = pending.front() {
362                    // Non-blocking: check if the oldest readback is ready
363                    if !worker.render_textures[prev].try_finish_readback(&worker.ctx) {
364                        break;
365                    }
366                    let (prev, prev_fc) = pending.pop_front().unwrap();
367                    worker.output_frame_from(prev, prev_fc);
368                }
369            }
370
371            // Flush all remaining pending frames
372            while let Some((prev, prev_fc)) = pending.pop_front() {
373                worker.render_textures[prev].finish_readback(&worker.ctx);
374                worker.output_frame_from(prev, prev_fc);
375            }
376
377            worker_tx.send_blocking(worker).unwrap();
378        });
379        RenderThreadHandle {
380            submit_frame_tx,
381            back_rx,
382            worker_rx,
383        }
384    }

386    /// Renders a single frame synchronously on this worker (typically used for captures).
387    pub fn render_store(&mut self, store: &RenderFrame) {
388        #[cfg(feature = "profiling")]
389        profiling::scope!("frame");
390
391        {
392            #[cfg(feature = "profiling")]
393            profiling::scope!("render");
394
395            self.renderer
396                .render_frame(&mut self.render_textures[0], self.clear_color, store);
397        }
398
399        #[cfg(feature = "profiling")]
400        profiling::finish_frame!();
401    }

403    /// Write and save (if [`Self::save_frames`] is true)
404    pub fn output_frame_from(&mut self, target_idx: usize, frame_number: u64) {
405        self.write_frame_from(target_idx);
406        if self.save_frames {
407            self.save_frame_from(target_idx, frame_number);
408        }
409    }

411    /// Write frame data from the given target to the video file.
412    pub fn write_frame_from(&mut self, target_idx: usize) {
413        let data = self.render_textures[target_idx]
414            .render_texture
415            .texture_data();
416        if let Some(video_writer) = self.video_writer.as_mut() {
417            video_writer.write_frame(data);
418        } else if let Some(builder) = self.video_writer_builder.as_ref() {
419            self.video_writer
420                .get_or_insert(builder.clone().build())
421                .write_frame(data);
422        }
423    }

425    /// Save frame from the given target as a PNG image.
426    pub fn save_frame_from(&mut self, target_idx: usize, frame_number: u64) {
427        let path = self.save_frame_dir().join(format!("{frame_number:04}.png"));
428        let dir = path.parent().unwrap();
429        if !dir.exists() || !dir.is_dir() {
430            std::fs::create_dir_all(dir).unwrap();
431        }
432        // Data is already in cpu buffer after finish_readback, this won't trigger GPU work
433        let buffer = self.render_textures[target_idx].get_rendered_texture_img_buffer(&self.ctx);
434        buffer.save(path).unwrap();
435    }
436
437    /// Capture frame to image file (sync path, uses target 0).
438    pub fn capture_frame(&mut self, path: impl AsRef<Path>) {
439        let path = path.as_ref();
440        let path = if !path.is_absolute() {
441            self.output_dir
442                .join(format!(
443                    "{}_{}x{}_{}",
444                    self.scene_name, self.width, self.height, self.fps
445                ))
446                .join(path)
447        } else {
448            path.to_path_buf()
449        };
450        let dir = path.parent().unwrap();
451        if !dir.exists() || !dir.is_dir() {
452            std::fs::create_dir_all(dir).unwrap();
453        }
454        let buffer = self.render_textures[0].get_rendered_texture_img_buffer(&self.ctx);
455        buffer.save(path).unwrap();
456    }

460/// Application wrapper that drives frame-by-frame scene rendering, video output, and capture screenshots.
461pub struct RanimRenderApp {
462    render_worker: Option<RenderWorker>,
463    fps: u32,
464    store: RenderFrame,
465}
466
467impl RanimRenderApp {
468    /// Creates the application and initializes the internal [`RenderWorker`].
469    pub fn new(
470        scene_name: String,
471        scene_config: &SceneConfig,
472        output: &Output,
473        buffer_count: usize,
474    ) -> Self {
475        let render_worker = RenderWorker::new(scene_name, scene_config, output, buffer_count);
476        Self {
477            render_worker: Some(render_worker),
478            fps: output.fps,
479            store: RenderFrame::default(),
480        }
481    }
482
483    /// Renders the entire scene and reports progress via a tracing progress bar and an optional `on_progress` callback.
484    ///
485    /// The frame count is `evaluator.total_secs() * fps` rounded up, with an extra frame appended when necessary to sample the final state.
486    #[instrument(skip_all)]
487    pub fn render_scene_with_progress(
488        &mut self,
489        evaluator: &mut SceneEvaluator,
490        on_progress: Option<Box<dyn Fn(u64, u64) + Send>>,
491    ) {
492        let start = Instant::now();
493        #[cfg(feature = "profiling")]
494        let (_cpu_server, _gpu_server) = {
495            puffin::set_scopes_on(true);
496            // default global profiler
497            let cpu_server =
498                puffin_http::Server::new(&format!("0.0.0.0:{}", puffin_http::DEFAULT_PORT))
499                    .unwrap();
500            // custom gpu profiler in `PUFFIN_GPU_PROFILER`
501            let gpu_server = puffin_http::Server::new_custom(
502                &format!("0.0.0.0:{}", puffin_http::DEFAULT_PORT + 1),
503                |sink| PUFFIN_GPU_PROFILER.lock().unwrap().add_sink(sink),
504                |id| _ = PUFFIN_GPU_PROFILER.lock().unwrap().remove_sink(id),
505            )
506            .unwrap();
507            (cpu_server, gpu_server)
508        };
509
510        let worker_thread = self.render_worker.take().unwrap().yeet();
511
512        let total_secs = evaluator.total_secs();
513        let fps = self.fps as f64;
514        let raw_frames = total_secs * fps;
515        // Add an extra frame to sample the final state exactly,
516        // unless total_secs * fps is already an integer (last frame lands on total_secs).
517        let n = raw_frames.ceil() as u64;
518        let num_frames = if (raw_frames - raw_frames.round()).abs() < 1e-9 {
519            n
520        } else {
521            n + 1
522        };
523        let style =             ProgressStyle::with_template(
524                "[{elapsed_precise}] [{wide_bar:.cyan/blue}] frame {human_pos}/{human_len} (eta {eta}) {msg}",
525            )
526            .unwrap()
527            .with_key("eta", |state: &ProgressState, w: &mut dyn std::fmt::Write| {
528                write!(w, "{:.1}s", state.eta().as_secs_f64()).unwrap()
529            })
530            .progress_chars("#>-");
531
532        let span = Span::current();
533        span.pb_set_style(&style);
534        span.pb_set_length(num_frames);
535
536        (0..num_frames)
537            .map(|f| (f as f64 / fps).min(total_secs))
538            .enumerate()
539            .for_each(|(i, sec)| {
540                let mut frame_items = Vec::new();
541                evaluator.sample_at(sec, &mut frame_items);
542                worker_thread.sync_and_submit(move |store| {
543                    store.update(frame_items.into_iter());
544                });
545
546                span.pb_inc(1);
547                if let Some(cb) = &on_progress {
548                    cb(i as u64 + 1, num_frames);
549                }
550                span.pb_set_message(
551                    format!(
552                        "rendering {:.1?}/{:.1?}",
553                        Duration::from_secs_f64(sec),
554                        Duration::from_secs_f64(total_secs)
555                    )
556                    .as_str(),
557                );
558            });
559        self.render_worker.replace(worker_thread.retrive());
560
561        info!(
562            "rendered {} frames({:?}) in {:?}",
563            num_frames,
564            Duration::from_secs_f64(evaluator.total_secs()),
565            start.elapsed(),
566        );
567        trace!("render timeline cost: {:?}", start.elapsed());
568    }
569
570    /// Renders and saves screenshots for every [`TimeMark::Capture`] mark on the timeline.
571    #[instrument(skip_all)]
572    pub fn render_capture_marks(&mut self, evaluator: &mut SceneEvaluator) {
573        let start = Instant::now();
574        let timemarks = evaluator
575            .time_marks()
576            .iter()
577            .filter(|mark| matches!(mark.1, TimeMark::Capture(_)))
578            .map(|(sec, mark)| (*sec, mark.clone()))
579            .collect::<Vec<_>>();
580
581        let style =             ProgressStyle::with_template(
582                "[{elapsed_precise}] [{wide_bar:.cyan/blue}] frame {human_pos}/{human_len} (eta {eta}) {msg}",
583            )
584            .unwrap()
585            .with_key("eta", |state: &ProgressState, w: &mut dyn std::fmt::Write| {
586                write!(w, "{:.1}s", state.eta().as_secs_f64()).unwrap()
587            })
588            .progress_chars("#>-");
589
590        let span = Span::current();
591        span.pb_set_style(&style);
592        span.pb_set_length(timemarks.len() as u64);
593        let _enter = span.enter();
594
595        let mut captured = 0usize;
596        for (sec, TimeMark::Capture(filename)) in timemarks {
597            // The render has advanced to the end; captures seek back and replay
598            // (deterministic contract, direction handled inside sample_at).
599            let mut frame_items = Vec::new();
600            evaluator.sample_at(sec, &mut frame_items);
601            self.store.update(frame_items.into_iter());
602            let worker = self.render_worker.as_mut().unwrap();
603            worker.render_store(&self.store);
604            worker.capture_frame(filename);
605            span.pb_inc(1);
606            captured += 1;
607        }
608        info!("saved {} capture frames from time marks", captured);
609
610        trace!("save capture frames cost: {:?}", start.elapsed());
611    }

627/// Download latest release of ffmpeg from <https://github.com/eugeneware/ffmpeg-static/releases/latest> to <target_dir>/ffmpeg
628pub fn download_ffmpeg(target_dir: impl AsRef<Path>) -> Result<PathBuf, anyhow::Error> {
629    use anyhow::Context;
630    use itertools::Itertools;
631    use std::io::Read;
632    use tracing::info;
633
634    let target_dir = target_dir.as_ref();
635
636    let res = reqwest::blocking::get(FFMPEG_RELEASE_URL).context("failed to get release url")?;
637    let url = res.url().to_string();
638    let url = url.split("tag").collect_array::<2>().unwrap();
639    let url = format!("{}/download/{}", url[0], url[1]);
640    info!("ffmpeg release url: {url:?}");
641
642    #[cfg(all(target_os = "windows", target_arch = "x86_64"))]
643    let url = format!("{url}/ffmpeg-win32-x64.gz");
644    #[cfg(all(target_os = "linux", target_arch = "x86_64"))]
645    let url = format!("{url}/ffmpeg-linux-x64.gz");
646    #[cfg(all(target_os = "linux", target_arch = "aarch64"))]
647    let url = format!("{url}/ffmpeg-linux-arm64.gz");
648    #[cfg(all(target_os = "macos", target_arch = "x86_64"))]
649    let url = format!("{url}/ffmpeg-darwin-x64.gz");
650    #[cfg(all(target_os = "macos", target_arch = "aarch64"))]
651    let url = format!("{url}/ffmpeg-darwin-arm64.gz");
652
653    info!("downloading ffmpeg from {url:?}...");
654
655    let res = reqwest::blocking::get(&url).context("get err")?;
656    let mut decoder = flate2::bufread::GzDecoder::new(std::io::BufReader::new(
657        std::io::Cursor::new(res.bytes().unwrap()),
658    ));
659    let mut bytes = Vec::new();
660    decoder
661        .read_to_end(&mut bytes)
662        .context("GzDecoder decode err")?;
663    let ffmpeg_path = target_dir.join("ffmpeg");
664    std::fs::write(&ffmpeg_path, bytes).unwrap();
665
666    #[cfg(target_family = "unix")]
667    {
668        use std::os::unix::fs::PermissionsExt;
669
670        std::fs::set_permissions(&ffmpeg_path, std::fs::Permissions::from_mode(0o755))?;
671    }
672    info!("ffmpeg downloaded to {target_dir:?}");
673    Ok(ffmpeg_path)
674}
