48pub struct Resolution {
49    pub width: u32,
50    pub height: u32,
51}

53impl Resolution {
54    pub const fn new(width: u32, height: u32) -> Self {
55        Self { width, height }
56    }
57
58    pub fn ratio(&self) -> f32 {
59        self.width as f32 / self.height as f32
60    }
61
62    /// Calculate and return the simplified aspect ratio (e.g., (16, 9) for 1920x1080)
63    pub fn aspect_ratio(&self) -> (u32, u32) {
64        fn gcd(a: u32, b: u32) -> u32 {
65            if b == 0 { a } else { gcd(b, a % b) }
66        }
67        let g = gcd(self.width, self.height);
68        (self.width / g, self.height / g)
69    }
70
71    pub fn aspect_ratio_str(&self) -> String {
72        let (w, h) = self.aspect_ratio();
73        format!("{w}:{h}")
74    }
75}

234    /// Calculate OIT layers based on resolution to stay within GPU buffer limits
235    fn calculate_oit_layers(&self, ctx: &WgpuContext, width: u32, height: u32) -> usize {
236        const BYTES_PER_PIXEL_PER_LAYER: usize = 8; // 4 bytes color + 4 bytes depth
237        const MAX_OIT_LAYERS: usize = 8;
238
239        let limits = ctx.device.limits();
240        let max_buffer_size = limits.max_storage_buffer_binding_size as usize;
241        let pixel_count = (width * height) as usize;
242        let max_layers_by_buffer = max_buffer_size / (pixel_count * BYTES_PER_PIXEL_PER_LAYER);
243        let oit_layers = max_layers_by_buffer.clamp(1, MAX_OIT_LAYERS);
244
245        if oit_layers < MAX_OIT_LAYERS {
246            tracing::warn!(
247                "OIT layers reduced from {} to {} due to GPU buffer size limit ({}MB @ {}x{})",
248                MAX_OIT_LAYERS,
249                oit_layers,
250                max_buffer_size / 1024 / 1024,
251                width,
252                height
253            );
254        }
255
256        oit_layers
257    }

259    fn handle_events(&mut self) {
260        if let Ok(cmd) = self.cmd_rx.try_recv() {
261            match cmd {
262                RanimPreviewAppCmd::ReloadScene(scene, tx) => {
263                    let timeline = scene.constructor.build_scene();
264                    let animation_infos = timeline.get_animation_infos();
265                    let old_cur_second = self.timeline_state.current_sec;
266                    self.timeline_state =
267                        TimelineState::new(timeline.total_secs(), animation_infos);
268                    self.timeline_state.current_sec =
269                        old_cur_second.clamp(0.0, self.timeline_state.total_sec);
270                    self.evaluator = timeline.into_evaluator(DEFAULT_LOGIC_FPS);
271                    self.store.update(std::iter::empty());
272                    self.need_eval = true;
273
274                    self.set_clear_color_str(&scene.config.clear_color);
275
276                    if let Err(err) = tx.try_send(()) {
277                        error!("Failed to send reloaded signal: {err:?}");
278                    }
279                }
280            }
281        }
282    }

284    fn prepare_renderer(&mut self, frame: &eframe::Frame) {
285        // Check if we need to recreate renderer
286        let needs_init = self.renderer.is_none();
287        let needs_resize = self.resolution_dirty && self.renderer.is_some();
288
289        if !needs_init && !needs_resize {
290            return;
291        }
292
293        let Some(render_state) = frame.wgpu_render_state() else {
294            tracing::info!("frame.wgpu_render_state() is none");
295            tracing::info!("{:?}", frame.info());
296            return;
297        };
298
299        if needs_init {
300            tracing::info!("preparing renderer...");
301        } else if needs_resize {
302            tracing::info!("recreating renderer for resolution change...");
303        }
304
305        // Construct WgpuContext using eframe's resources.
306        // NOTE: We assume ranim-render doesn't strictly depend on the instance for the operations we do here.
307        let ctx = WgpuContext {
308            instance: wgpu::Instance::default(), // Dummy instance
309            adapter: wgpu::Adapter::clone(&render_state.adapter),
310            device: wgpu::Device::clone(&render_state.device),
311            queue: wgpu::Queue::clone(&render_state.queue),
312        };
313
314        let (width, height) = (self.resolution.width, self.resolution.height);
315        let oit_layers = self.calculate_oit_layers(&ctx, width, height);
316        let renderer = Renderer::new(&ctx, width, height, oit_layers);
317        let render_textures = renderer.new_render_textures(&ctx);
318
319        // Init Depth Visual Pipeline
320        if self.depth_visual_pipeline.is_none() {
321            self.depth_visual_pipeline = Some(DepthVisualPipeline::new(&ctx));
322        }
323
324        // Create Depth Visual Texture
325        let depth_visual_texture = ctx.device.create_texture(&wgpu::TextureDescriptor {
326            label: Some("Depth Visual Texture"),
327            size: wgpu::Extent3d {
328                width: render_textures.width(),
329                height: render_textures.height(),
330                depth_or_array_layers: 1,
331            },
332            mip_level_count: 1,
333            sample_count: 1,
334            dimension: wgpu::TextureDimension::D2,
335            format: wgpu::TextureFormat::Rgba8Unorm,
336            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
337            view_formats: &[],
338        });
339        let depth_visual_view =
340            depth_visual_texture.create_view(&wgpu::TextureViewDescriptor::default());
341
342        // Register texture with egui
343        let texture_view = &render_textures.linear_render_view;
344        let texture_id = render_state.renderer.write().register_native_texture(
345            &render_state.device,
346            texture_view,
347            wgpu::FilterMode::Linear,
348        );
349        let depth_id = render_state.renderer.write().register_native_texture(
350            &render_state.device,
351            &depth_visual_view,
352            wgpu::FilterMode::Nearest,
353        );
354
355        self.texture_id = Some(texture_id);
356        self.depth_texture_id = Some(depth_id);
357        self.depth_visual_texture = Some(depth_visual_texture);
358        self.depth_visual_view = Some(depth_visual_view);
359        self.render_textures = Some(render_textures);
360        self.renderer = Some(renderer);
361        self.wgpu_ctx = Some(ctx);
362        self.resolution_dirty = false;
363        self.need_eval = true; // Force re-render with new resolution
364    }

366    fn render_animation(&mut self) {
367        if let (Some(ctx), Some(renderer), Some(render_textures)) = (
368            self.wgpu_ctx.as_ref(),
369            self.renderer.as_mut(),
370            self.render_textures.as_mut(),
371        ) {
372            if self.last_sec == self.timeline_state.current_sec && !self.need_eval {
373                return;
374            }
375            self.need_eval = false;
376            self.last_sec = self.timeline_state.current_sec;
377
378            let start_eval = Instant::now();
379            // Forward/backward direction management is internal to sample_at.
380            let target = self.timeline_state.current_sec;
381            let mut frame_items = Vec::new();
382            self.evaluator.sample_at(target, &mut frame_items);
383            self.store.update(frame_items.into_iter());
384            self.last_eval_time = Some(start_eval.elapsed());
385
386            let start = Instant::now();
387            renderer.render_frame(render_textures, self.clear_color, &self.store);
388
389            if let (Some(pipeline), Some(view)) = (
390                self.depth_visual_pipeline.as_ref(),
391                self.depth_visual_view.as_ref(),
392            ) {
393                let mut encoder =
394                    ctx.device
395                        .create_command_encoder(&wgpu::CommandEncoderDescriptor {
396                            label: Some("Depth Visual Encoder"),
397                        });
398
399                let bind_group = ctx.device.create_bind_group(&wgpu::BindGroupDescriptor {
400                    label: Some("Depth Visual Bind Group"),
401                    layout: &pipeline.bind_group_layout,
402                    entries: &[wgpu::BindGroupEntry {
403                        binding: 0,
404                        resource: wgpu::BindingResource::TextureView(
405                            &render_textures.depth_texture_view,
406                        ),
407                    }],
408                });
409
410                {
411                    let mut rpass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
412                        label: Some("Depth Visual Pass"),
413                        color_attachments: &[Some(wgpu::RenderPassColorAttachment {
414                            view,
415                            resolve_target: None,
416                            depth_slice: None,
417                            ops: wgpu::Operations {
418                                load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
419                                store: wgpu::StoreOp::Store,
420                            },
421                        })],
422                        depth_stencil_attachment: None,
423                        timestamp_writes: None,
424                        occlusion_query_set: None,
425                        multiview_mask: None,
426                    });
427                    rpass.set_pipeline(&pipeline.pipeline);
428                    rpass.set_bind_group(0, &bind_group, &[]);
429                    rpass.draw(0..3, 0..1);
430                }
431                ctx.queue.submit(Some(encoder.finish()));
432            }
433
434            self.last_render_time = Some(start.elapsed());
435        }
436    }

438    #[cfg(all(not(target_family = "wasm"), feature = "render"))]
439    fn start_export(&mut self, ctx: egui::Context) {
440        let (progress_tx, progress_rx) = unbounded();
441        self.export_progress_rx = Some(progress_rx);
442
443        let constructor = self.scene_constructor.clone();
444        let scene_config = self.scene_config.clone();
445        let output = self.export_config.clone();
446        let name = self.title.clone();
447
448        std::thread::spawn(move || {
449            let progress_tx_cb = progress_tx.clone();
450            let ctx_cb = ctx.clone();
451            let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
452                crate::cmd::render::render_scene_output_with_progress(
453                    constructor,
454                    name,
455                    &scene_config,
456                    &output,
457                    2,
458                    Some(Box::new(move |current, total| {
459                        let _ =
460                            progress_tx_cb.send_blocking(ExportProgress::Progress(current, total));
461                        ctx_cb.request_repaint();
462                    })),
463                );
464
465                let _ = progress_tx.send_blocking(ExportProgress::Done);
466                ctx.request_repaint();
467            }));
468
469            if let Err(e) = result {
470                let msg = if let Some(s) = e.downcast_ref::<&str>() {
471                    s.to_string()
472                } else if let Some(s) = e.downcast_ref::<String>() {
473                    s.clone()
474                } else {
475                    "Unknown export error".to_string()
476                };
477                let _ = progress_tx.send_blocking(ExportProgress::Error(msg));
478                ctx.request_repaint();
479            }
480        });
481    }

485    fn ui(&mut self, ui: &mut egui::Ui, frame: &mut eframe::Frame) {
486        let ctx = ui.ctx().clone();
487        self.prepare_renderer(frame);
488        self.handle_events();
489
490        // Space bar toggles play/pause
491        if ctx.input(|i| i.key_pressed(egui::Key::Space)) {
492            if self.play_prev_t.is_some() {
493                self.play_prev_t = None;
494            } else {
495                if self.timeline_state.current_sec >= self.timeline_state.total_sec {
496                    self.timeline_state.current_sec = 0.0;
497                }
498                self.play_prev_t = Some(Instant::now());
499            }
500        }
501
502        // Arrow keys step forward/back one frame
503        {
504            let frame_dur = 1.0 / self.export_config.fps as f64;
505            if ctx.input(|i| i.key_pressed(egui::Key::ArrowLeft)) {
506                self.play_prev_t = None;
507                self.timeline_state.current_sec =
508                    (self.timeline_state.current_sec - frame_dur).max(0.0);
509            }
510            if ctx.input(|i| i.key_pressed(egui::Key::ArrowRight)) {
511                self.play_prev_t = None;
512                self.timeline_state.current_sec = (self.timeline_state.current_sec + frame_dur)
513                    .min(self.timeline_state.total_sec);
514            }
515        }
516
517        if let Some(play_prev_t) = self.play_prev_t {
518            let elapsed = play_prev_t.elapsed().as_secs_f64() * self.playback_speed;
519            self.timeline_state.current_sec =
520                (self.timeline_state.current_sec + elapsed).min(self.timeline_state.total_sec);
521            if self.timeline_state.current_sec >= self.timeline_state.total_sec {
522                if self.looping {
523                    self.timeline_state.current_sec = 0.0;
524                    self.play_prev_t = Some(Instant::now());

650                        let exporting = self.export_progress_rx.is_some();
651                        if ui
652                            .add_enabled(!exporting, egui::Button::new("Export"))
653                            .clicked()
654                        {
655                            self.export_dialog_open = true;
656                        }
657                        ui.separator();
658                    }
659                    ui.selectable_value(&mut self.view_mode, ViewMode::Output, "Output");
660                    ui.selectable_value(&mut self.view_mode, ViewMode::Depth, "Depth");
661                    ui.separator();
662
663                    if let Some(duration) = self.last_render_time {
664                        ui.label(format!("Render: {:.2}ms", duration.as_secs_f64() * 1000.0));
665                        ui.separator();
666                    }
667                    if let Some(duration) = self.last_eval_time {
668                        ui.label(format!("Eval: {:.2}ms", duration.as_secs_f64() * 1000.0));
669                        ui.separator();
670                    }
671                });
672            });
673        });
674
675        egui::Panel::bottom("bottom_panel")
676            .resizable(true)
677            .default_size(240.0)
678            .min_size(150.0)
679            .max_size(600.0)
680            .show(ui, |ui| {
681                ui.label("Timeline");
682
683                ui.horizontal(|ui| {
684                    let fps = self.export_config.fps as f64;
685                    let frame_dur = 1.0 / fps;
686
687                    // |< Jump to start
688                    if ui
689                        .button(egui_phosphor::regular::SKIP_BACK)
690                        .on_hover_text("Jump to start")
691                        .clicked()
692                    {
693                        self.timeline_state.current_sec = 0.0;
694                        self.play_prev_t = None;
695                    }
696
697                    // < Step back one frame
698                    if ui
699                        .button(egui_phosphor::regular::CARET_LEFT)
700                        .on_hover_text("Step back one frame")
701                        .clicked()
702                    {
703                        self.play_prev_t = None;
704                        self.timeline_state.current_sec =
705                            (self.timeline_state.current_sec - frame_dur).max(0.0);
706                    }
707
708                    // Play / Pause
709                    let is_playing = self.play_prev_t.is_some();
710                    let play_label = if is_playing {
711                        egui_phosphor::regular::PAUSE
712                    } else {
713                        egui_phosphor::regular::PLAY
714                    };
715                    let play_tooltip = if is_playing { "Pause" } else { "Play" };
716                    if ui.button(play_label).on_hover_text(play_tooltip).clicked() {
717                        if is_playing {
718                            self.play_prev_t = None;
719                        } else {
720                            if self.timeline_state.current_sec >= self.timeline_state.total_sec {
721                                self.timeline_state.current_sec = 0.0;
722                            }
723                            self.play_prev_t = Some(Instant::now());
724                        }
725                    }
726
727                    // > Step forward one frame
728                    if ui
729                        .button(egui_phosphor::regular::CARET_RIGHT)
730                        .on_hover_text("Step forward one frame")
731                        .clicked()
732                    {
733                        self.play_prev_t = None;
734                        self.timeline_state.current_sec = (self.timeline_state.current_sec
735                            + frame_dur)
736                            .min(self.timeline_state.total_sec);
737                    }
738
739                    // >| Jump to end
740                    if ui
741                        .button(egui_phosphor::regular::SKIP_FORWARD)
742                        .on_hover_text("Jump to end")
743                        .clicked()
744                    {
745                        self.timeline_state.current_sec = self.timeline_state.total_sec;
746                        self.play_prev_t = None;
747                    }
748
749                    ui.separator();
750
751                    // Loop toggle
752                    let mut loop_btn = egui::Button::new(egui_phosphor::regular::ARROWS_CLOCKWISE);
753                    if self.looping {
754                        loop_btn = loop_btn.fill(ui.visuals().selection.bg_fill);
755                    }
756                    if ui
757                        .add(loop_btn)
758                        .on_hover_text(if self.looping {
759                            "Looping: ON"
760                        } else {
761                            "Looping: OFF"
762                        })
763                        .clicked()
764                    {
765                        self.looping = !self.looping;
766                    }
767
768                    ui.separator();
769
770                    // Speed control
771                    let drag_speed = (self.playback_speed * 0.02).max(0.01);
772                    ui.add(
773                        egui::DragValue::new(&mut self.playback_speed)
774                            .speed(drag_speed)
775                            .range(0.1..=10.0)
776                            .suffix("x"),
777                    )
778                    .on_hover_text("Playback speed");
779
780                    ui.separator();
781
782                    const TIME_INPUT_WIDTH: f32 = 86.0;
783                    let slider_width =
784                        (ui.available_width() - TIME_INPUT_WIDTH - ui.spacing().item_spacing.x)
785                            .max(40.0);
786                    ui.scope(|ui| {
787                        ui.style_mut().spacing.slider_width = slider_width;
788                        ui.add(
789                            egui::Slider::new(
790                                &mut self.timeline_state.current_sec,
791                                0.0..=self.timeline_state.total_sec,
792                            )
793                            .show_value(false),
794                        )
795                        .on_hover_text("Timeline position");
796                    });
797                    ui.add_sized(
798                        [TIME_INPUT_WIDTH, 18.0],
799                        egui::DragValue::new(&mut self.timeline_state.current_sec)
800                            .range(0.0..=self.timeline_state.total_sec)
801                            .speed(0.001)
802                            .fixed_decimals(3)
803                            .suffix(" s"),
804                    )
805                    .on_hover_text("Drag or enter the current time");
806                });
807
808                self.timeline_state.ui_main_timeline(ui);
809            });
810
811        egui::CentralPanel::default().show(ui, |ui| {
812            let texture_id = match self.view_mode {
813                ViewMode::Output => self.texture_id,
814                ViewMode::Depth => self.depth_texture_id,
815            };
816
817            if let Some(tid) = texture_id {
818                // Maintain aspect ratio
819                // TODO: We could update renderer size here if we want dynamic resolution
820                let available_size = ui.available_size();
821                let aspect_ratio = self
822                    .render_textures
823                    .as_ref()
824                    .map(|rt| rt.ratio())
825                    .unwrap_or(1280.0 / 7.0);
826                let mut size = available_size;
827
828                if size.x / size.y > aspect_ratio {
829                    size.x = size.y * aspect_ratio;
830                } else {
831                    size.y = size.x / aspect_ratio;
832                }
833
834                ui.centered_and_justified(|ui| {
835                    ui.image(egui::load::SizedTexture::new(tid, size));
836                });
837            } else {
838                ui.centered_and_justified(|ui| {
839                    ui.spinner();
840                });
841            }
842        });
843
844        // Export (native only)
845        #[cfg(all(not(target_family = "wasm"), feature = "render"))]
846        {
847            // Poll export progress
848            if let Some(rx) = &self.export_progress_rx {
849                let mut done = false;
850                let mut error_msg = None;
851
852                while let Ok(msg) = rx.try_recv() {
853                    match msg {
854                        ExportProgress::Progress(current, total) => {
855                            self.export_current_frame = current;
856                            self.export_total_frames = total;
857                        }
858                        ExportProgress::Done => {
859                            done = true;
860                        }
861                        ExportProgress::Error(err) => {
862                            error_msg = Some(err);
863                            done = true;
864                        }
865                    }
866                }
867
868                if done {
869                    self.export_progress_rx = None;
870                    self.export_current_frame = 0;
871                    self.export_total_frames = 0;
872                    if let Some(err) = error_msg {
873                        error!("Export failed: {err}");
874                    } else {
875                        info!("Export completed");
876                    }
877                } else {
878                    ctx.request_repaint();
879                }
880            }
881
882            // Export configuration dialog
883            let exporting = self.export_progress_rx.is_some();
884            if self.export_dialog_open || exporting {
885                let mut open = self.export_dialog_open;
886                egui::Window::new("Export")
887                    .open(&mut open)
888                    .resizable(false)
889                    .show(&ctx, |ui| {
890                        ui.add_enabled_ui(!exporting, |ui| {
891                            egui::Grid::new("export_grid")
892                                .num_columns(2)
893                                .show(ui, |ui| {
894                                    ui.label("Width:");
895                                    ui.add(
896                                        egui::DragValue::new(&mut self.export_config.width)
897                                            .range(1..=7680),
898                                    );
899                                    ui.end_row();
900
901                                    ui.label("Height:");
902                                    ui.add(
903                                        egui::DragValue::new(&mut self.export_config.height)
904                                            .range(1..=4320),
905                                    );
906                                    ui.end_row();
907
908                                    ui.label("FPS:");
909                                    ui.add(
910                                        egui::DragValue::new(&mut self.export_config.fps)
911                                            .range(1..=240),
912                                    );
913                                    ui.end_row();
914
915                                    ui.label("Format:");
916                                    egui::ComboBox::from_id_salt("export_format")
917                                        .selected_text(format!("{}", self.export_config.format))
918                                        .show_ui(ui, |ui| {
919                                            ui.selectable_value(
920                                                &mut self.export_config.format,
921                                                OutputFormat::Mp4,
922                                                "mp4",
923                                            );
924                                            ui.selectable_value(
925                                                &mut self.export_config.format,
926                                                OutputFormat::Webm,
927                                                "webm",
928                                            );
929                                            ui.selectable_value(
930                                                &mut self.export_config.format,
931                                                OutputFormat::Mov,
932                                                "mov",
933                                            );
934                                            ui.selectable_value(
935                                                &mut self.export_config.format,
936                                                OutputFormat::Gif,
937                                                "gif",
938                                            );
939                                        });
940                                    ui.end_row();
941
942                                    ui.label("Output dir:");
943                                    ui.text_edit_singleline(&mut self.export_config.dir);
944                                    ui.end_row();
945
946                                    // Show resolved output path preview right below the dir input
947                                    ui.label("");
948                                    {
949                                        let mut output_dir =
950                                            std::path::PathBuf::from(&self.export_config.dir);
951                                        if !output_dir.is_absolute() {
952                                            output_dir = std::env::current_dir()
953                                                .unwrap_or_default()
954                                                .join(&output_dir);
955                                        }
956                                        let (_, _, ext) =
957                                            self.export_config.format.encoding_params();
958                                        let name = self
959                                            .export_config
960                                            .name
961                                            .as_deref()
962                                            .unwrap_or(&self.title);
963                                        let file_path = output_dir.join(format!(
964                                            "{}_{}x{}_{}.{ext}",
965                                            name,
966                                            self.export_config.width,
967                                            self.export_config.height,
968                                            self.export_config.fps,
969                                        ));
970                                        ui.label(
971                                            egui::RichText::new(format!(
972                                                "-> {}",
973                                                file_path.display()
974                                            ))
975                                            .small()
976                                            .color(ui.visuals().weak_text_color()),
977                                        );
978                                    }
979                                    ui.end_row();
980
981                                    ui.label("Save frames:");
982                                    ui.checkbox(&mut self.export_config.save_frames, "");
983                                    ui.end_row();
984                                });
985                        }); // end add_enabled_ui
986
987                        ui.add_space(8.0);
988
989                        // Show progress bar inline when exporting
990                        if exporting {
991                            let current = self.export_current_frame;
992                            let total = self.export_total_frames;
993                            if total > 0 {
994                                let progress = current as f32 / total as f32;
995                                ui.add(egui::ProgressBar::new(progress).text(format!(
996                                    "{current}/{total} frames ({:.0}%)",
997                                    progress * 100.0
998                                )));
999                            } else {
1000                                ui.horizontal(|ui| {
1001                                    ui.spinner();
1002                                    ui.label("Preparing...");
1003                                });
1004                            }
1005                        } else if ui.button("Start Export").clicked() {
1006                            self.start_export(ctx.clone());
1007                        }
1008                    });
1009                // Don't allow closing the window while exporting
1010                if !exporting {
1011                    self.export_dialog_open = open;
1012                }
1013            }
1014        }
1015    }

1018pub fn run_app(app: RanimPreviewApp, #[cfg(target_arch = "wasm32")] container_id: String) {
1019    #[cfg(not(target_arch = "wasm32"))]
1020    let title = app.title.clone();
1021    let build_app = |cc: &eframe::CreationContext| {
1022        let mut fonts = egui::FontDefinitions::default();
1023        egui_phosphor::add_to_fonts(&mut fonts, egui_phosphor::Variant::Regular);
1024        cc.egui_ctx.set_fonts(fonts);
1025        Ok(Box::new(app) as Box<dyn App>)
1026    };
1027
1028    // NOTE: eframe defaults to wgpu's default limits (8 storage buffers per shader
1029    // stage), but our pipelines exceed that (the VItem pipelines bind 9 storage
1030    // buffers in the fragment stage). Request the adapter's full limits, matching
1031    // the offline `WgpuContext::new()`, or pipeline creation panics in preview.
1032    let wgpu_options = eframe::egui_wgpu::WgpuConfiguration {
1033        wgpu_setup: eframe::egui_wgpu::WgpuSetup::CreateNew(
1034            eframe::egui_wgpu::WgpuSetupCreateNew {
1035                device_descriptor: Arc::new(|adapter| wgpu::DeviceDescriptor {
1036                    label: Some("ranim device"),
1037                    required_limits: adapter.limits(),
1038                    ..Default::default()
1039                }),
1040                ..eframe::egui_wgpu::WgpuSetupCreateNew::without_display_handle()
1041            },
1042        ),
1043        ..Default::default()
1044    };
1045
1046    #[cfg(not(target_family = "wasm"))]
1047    {
1048        let native_options = eframe::NativeOptions {
1049            viewport: egui::ViewportBuilder::default()
1050                .with_title(&title)
1051                .with_inner_size([1280.0, 720.0]),
1052            renderer: eframe::Renderer::Wgpu,
1053            wgpu_options,
1054            ..Default::default()
1055        };
1056
1057        // We need to clone title because run_native takes String (or &str) and app is moved into closure
1058
1059        eframe::run_native(&title, native_options, Box::new(build_app)).unwrap();
1060    }
1061
1062    #[cfg(target_arch = "wasm32")]
1063    {
1064        use wasm_bindgen::JsCast;
1065        let web_options = eframe::WebOptions {
1066            wgpu_options,
1067            ..Default::default()
1068        };
1069
1070        // Handling canvas creation if not found to ensure compatibility
1071        let document = web_sys::window().unwrap().document().unwrap();
1072        let canvas = document
1073            .get_element_by_id(&container_id)
1074            .and_then(|c| c.dyn_into::<web_sys::HtmlCanvasElement>().ok());
1075
1076        let canvas = if let Some(canvas) = canvas {
1077            canvas
1078        } else {
1079            let canvas = document.create_element("canvas").unwrap();
1080            canvas.set_id(&container_id);
1081            document.body().unwrap().append_child(&canvas).unwrap();
1082            canvas.dyn_into::<web_sys::HtmlCanvasElement>().unwrap()
1083        };
1084
1085        wasm_bindgen_futures::spawn_local(async {
1086            eframe::WebRunner::new()
1087                .start(canvas, web_options, Box::new(build_app))
1088                .await
1089                .expect("failed to start eframe");
1090        });
1091    }
1092}

1096    name: &str,
