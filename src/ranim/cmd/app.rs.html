17struct SceneSnapshot {
18    scene: RanimScene,
19    build_time: Duration,
20}
21
22impl SceneSnapshot {
23    fn build(scene: &Scene) -> Result<Self, String> {
24        catch_unwind(AssertUnwindSafe(|| {
25            let start = Instant::now();
26            let definition = scene.build_definition();
27            Self {
28                scene: definition,
29                build_time: start.elapsed(),
30            }
31        }))
32        .map_err(|payload| {
33            payload
34                .downcast_ref::<String>()
35                .cloned()
36                .or_else(|| payload.downcast_ref::<&str>().map(|s| (*s).to_owned()))
37                .unwrap_or_else(|| "Scene constructor panicked with a non-string payload".into())
38        })
39    }
40}
41
42fn count_nodes(roots: &[AnimDefNode]) -> usize {
43    roots
44        .iter()
45        .map(|node| 1 + count_nodes(node.definitions()))
46        .sum()
47}
48
49fn node_at_path<'a>(roots: &'a [AnimDefNode], path: &[usize]) -> Option<&'a AnimDefNode> {
50    let (index, rest) = path.split_first()?;
51    let node = roots.get(*index)?;
52    if rest.is_empty() {
53        Some(node)
54    } else {
55        node_at_path(node.definitions(), rest)
56    }
57}
58
59fn short_name(name: &str) -> &str {
60    name.split('<')
61        .next()
62        .unwrap_or(name)
63        .rsplit("::")
64        .next()
65        .unwrap_or(name)
66}
67
68fn kind_name(node: &AnimDefNode) -> &'static str {
69    if node.is_container() {
70        "Container"
71    } else if node.is_audio() {
72        "Audio"
73    } else {
74        "Animation"
75    }
76}

78/// An inspector for the scenes and retained animation definitions of an assembled App.
79pub struct InspectorApp {
80    // Keep project resources alive for captured scene constructors.
81    _project: RanimApp,
82    scenes: Vec<Scene>,
83    filter: String,
84    selected_scene: Option<usize>,
85    selected_node: Vec<usize>,
86    snapshots: BTreeMap<usize, Result<SceneSnapshot, String>>,
87    expand_all: Option<bool>,
88}
89
90impl InspectorApp {
91    /// Create an inspector without building any scenes or initializing a renderer.
92    pub fn new(project: RanimApp) -> Self {
93        let scenes = project.scenes().to_vec();
94        Self {
95            _project: project,
96            scenes,
97            filter: String::new(),
98            selected_scene: None,
99            selected_node: Vec::new(),
100            snapshots: BTreeMap::new(),
101            expand_all: None,
102        }
103    }
104
105    fn build_selected_scene(&mut self) {
106        if let Some(index) = self.selected_scene
107            && !self.snapshots.contains_key(&index)
108        {
109            self.snapshots
110                .insert(index, SceneSnapshot::build(&self.scenes[index]));
111        }
112    }
113
114    fn show(&mut self, ui: &mut egui::Ui) {
115        self.build_selected_scene();
116        egui::Panel::top("inspector-toolbar").show(ui, |ui| {
117            ui.horizontal(|ui| {
118                ui.heading("Ranim Inspector");
119                ui.separator();
120                ui.label(format!("{} scenes", self.scenes.len()));
121                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
122                    ui.weak("Read-only");
123                });
124            });
125        });
126
127        egui::Panel::left("inspector-scenes")
128            .default_size(230.0)
129            .resizable(true)
130            .show(ui, |ui| {
131                ui.heading("Scenes");
132                ui.add(egui::TextEdit::singleline(&mut self.filter).hint_text("Filter scenes…"));
133                ui.separator();
134                let filter = self.filter.to_lowercase();
135                egui::ScrollArea::vertical()
136                    .id_salt("scene-list")
137                    .show(ui, |ui| {
138                        for (index, scene) in self.scenes.iter().enumerate() {
139                            if !scene.name.to_lowercase().contains(&filter) {
140                                continue;
141                            }
142                            if ui
143                                .selectable_label(self.selected_scene == Some(index), &scene.name)
144                                .clicked()
145                            {
146                                self.selected_scene = Some(index);
147                                self.selected_node.clear();
148                                ui.ctx().request_repaint();
149                            }
150                        }
151                        if self.scenes.is_empty() {
152                            ui.weak("No scenes registered in this App.");
153                        }
154                    });
155            });
156
157        egui::Panel::right("inspector-properties")
158            .default_size(340.0)
159            .min_size(230.0)
160            .resizable(true)
161            .show(ui, |ui| {
162                ui.heading("Inspector");
163                egui::ScrollArea::vertical()
164                    .id_salt("properties")
165                    .show(ui, |ui| {
166                        let node = self
167                            .selected_scene
168                            .and_then(|index| self.snapshots.get(&index))
169                            .and_then(|result| result.as_ref().ok())
170                            .and_then(|snapshot| {
171                                node_at_path(snapshot.scene.root.definitions(), &self.selected_node)
172                            });
173                        if let Some(node) = node {
174                            show_node_details(ui, node, &self.selected_node);
175                        } else {
176                            ui.weak("Select an animation node to inspect it.");
177                        }
178                    });
179            });
180
181        egui::CentralPanel::default().show(ui, |ui| {
182            let Some(index) = self.selected_scene else {
183                ui.centered_and_justified(|ui| {
184                    ui.weak(
185                        "Select a scene from the left to build and inspect its animation tree.",
186                    );
187                });
188                return;
189            };
190            ui.heading(&self.scenes[index].name);
191            ui.horizontal(|ui| {
192                if ui.button("Rebuild scene").clicked() {
193                    self.snapshots.remove(&index);
194                    self.selected_node.clear();
195                    ui.ctx().request_repaint();
196                }
197                if ui.button("Expand all").clicked() {
198                    self.expand_all = Some(true);
199                }
200                if ui.button("Collapse all").clicked() {
201                    self.expand_all = Some(false);
202                }
203            });
204            ui.separator();
205            match self.snapshots.get(&index) {
206                Some(Ok(snapshot)) => {
207                    ui.label(format!(
208                        "{:.3} s · {} root nodes · {} total nodes · built in {:.2?}",
209                        snapshot.scene.root.duration_secs(),
210                        snapshot.scene.root.definitions().len(),
211                        count_nodes(snapshot.scene.root.definitions()),
212                        snapshot.build_time
213                    ));
214                    ui.weak(
215                        "Ranges are in the parent's local content time, not global scene time.",
216                    );
217                    egui::ScrollArea::both()
218                        .id_salt(("animation-tree", index))
219                        .show(ui, |ui| {
220                            if let Some(open) = self.expand_all {
221                                set_descendant_expansion(
222                                    ui,
223                                    snapshot.scene.root.definitions(),
224                                    &mut Vec::new(),
225                                    index,
226                                    open,
227                                );
228                            }
229                            show_tree(
230                                ui,
231                                snapshot.scene.root.definitions(),
232                                &mut Vec::new(),
233                                &mut self.selected_node,
234                                index,
235                            );
236                        });
237                    if snapshot.scene.root.definitions().is_empty() {
238                        ui.weak("This scene has no animation nodes.");
239                    }
240                }
241                Some(Err(error)) => {
242                    ui.colored_label(ui.visuals().error_fg_color, "Scene build failed");
243                    ui.label(error);
244                    ui.weak("Other scenes remain available. Fix the scene and rebuild to retry.");
245                }
246                None => {
247                    ui.horizontal(|ui| {
248                        ui.spinner();
249                        ui.label("Waiting for scene construction…");
250                    });
251                }
252            }
253            self.expand_all = None;
254        });
255    }
256}

264fn show_tree(
265    ui: &mut egui::Ui,
266    nodes: &[AnimDefNode],
267    path: &mut Vec<usize>,
268    selected: &mut Vec<usize>,
269    scene: usize,
270) {
271    for (index, node) in nodes.iter().enumerate() {
272        path.push(index);
273        let range = node.time_range();
274        let label = format!(
275            "{}  [{:.3} … {:.3}]{}",
276            short_name(node.anim_name()),
277            range.start,
278            range.end,
279            if node.enabled() { "" } else { " · disabled" }
280        );
281        if node.definitions().is_empty() {
282            if ui
283                .selectable_label(*selected == *path, label)
284                .on_hover_text(node.anim_name())
285                .clicked()
286            {
287                *selected = path.clone();
288            }
289        } else {
290            let id = egui::Id::new(("animation-node", scene, &*path));
291            let state = egui::collapsing_header::CollapsingState::load_with_default_open(
292                ui.ctx(),
293                id,
294                path.len() < 3,
295            );
296            state
297                .show_header(ui, |ui| {
298                    if ui
299                        .selectable_label(*selected == *path, label)
300                        .on_hover_text(node.anim_name())
301                        .clicked()
302                    {
303                        *selected = path.clone();
304                    }
305                })
306                .body(|ui| {
307                    show_tree(ui, node.definitions(), path, selected, scene);
308                });
309        }
310        path.pop();
311    }
312}

314fn set_descendant_expansion(
315    ui: &egui::Ui,
316    nodes: &[AnimDefNode],
317    path: &mut Vec<usize>,
318    scene: usize,
319    open: bool,
320) {
321    for (index, node) in nodes.iter().enumerate() {
322        path.push(index);
323        let id = egui::Id::new(("animation-node", scene, &*path));
324        let mut state =
325            egui::collapsing_header::CollapsingState::load_with_default_open(ui.ctx(), id, open);
326        state.set_open(open);
327        state.store(ui.ctx());
328        set_descendant_expansion(ui, node.definitions(), path, scene, open);
329        path.pop();
330    }
331}

339fn show_node_details(ui: &mut egui::Ui, node: &AnimDefNode, path: &[usize]) {
340    let range = node.time_range();
341    ui.heading(short_name(node.anim_name()));
342    ui.label(egui::RichText::new(node.anim_name()).monospace().small());
343    ui.separator();
344    egui::Grid::new("node-properties")
345        .num_columns(2)
346        .striped(true)
347        .show(ui, |ui| {
348            property(
349                ui,
350                "Path",
351                path.iter()
352                    .map(usize::to_string)
353                    .collect::<Vec<_>>()
354                    .join("/"),
355            );
356            property(ui, "Kind", kind_name(node));
357            property(ui, "Enabled", node.enabled().to_string());
358            property(ui, "Start (parent s)", format!("{:.6}", range.start));
359            property(ui, "End (parent s)", format!("{:.6}", range.end));
360            property(ui, "Duration (s)", format!("{:.6}", node.duration_secs()));
361            property(
362                ui,
363                "Content duration (s)",
364                format!("{:.6}", node.content_duration_secs()),
365            );
366            property(ui, "Children", node.definitions().len().to_string());
367            property(
368                ui,
369                "Descendants",
370                count_nodes(node.definitions()).to_string(),
371            );
372            property(
373                ui,
374                "Simulation step",
375                node.sim_step()
376                    .map_or_else(|| "—".into(), |step| format!("{step:.8} progress")),
377            );
378        });
379    ui.separator();
380    ui.label("Rate function · local progress → content progress");
381    show_rate_curve(ui, node.rate_func());
382}

384fn show_rate_curve(ui: &mut egui::Ui, rate: fn(f64) -> f64) {
385    let (rect, _) = ui.allocate_exact_size(
386        egui::vec2(ui.available_width(), 120.0),
387        egui::Sense::hover(),
388    );
389    let painter = ui.painter_at(rect);
390    painter.rect_filled(rect, 4.0, ui.visuals().extreme_bg_color);
391    let plot = rect.shrink(8.0);
392    let points: Result<Vec<_>, _> = catch_unwind(|| {
393        (0..=64)
394            .map(|index| {
395                let x = f64::from(index) / 64.0;
396                (x, rate(x))
397            })
398            .collect()
399    });
400    let Ok(points) = points else {
401        painter.text(
402            rect.center(),
403            egui::Align2::CENTER_CENTER,
404            "Rate function panicked",
405            egui::FontId::default(),
406            ui.visuals().error_fg_color,
407        );
408        return;
409    };
410    let min = points
411        .iter()
412        .filter_map(|(_, y)| y.is_finite().then_some(*y))
413        .fold(0.0, f64::min);
414    let max = points
415        .iter()
416        .filter_map(|(_, y)| y.is_finite().then_some(*y))
417        .fold(1.0, f64::max);
418    let mut previous = None;
419    for (x, y) in points {
420        if !y.is_finite() {
421            previous = None;
422            continue;
423        }
424        let point = egui::pos2(
425            plot.left() + plot.width() * x as f32,
426            plot.bottom() - plot.height() * ((y - min) / (max - min)) as f32,
427        );
428        if let Some(previous) = previous {
429            painter.line_segment(
430                [previous, point],
431                egui::Stroke::new(1.5, ui.visuals().selection.bg_fill),
432            );
433        }
434        previous = Some(point);
435    }

438/// Launch a native egui inspector for all scenes registered in the project.
439pub fn run_app(app: RanimApp) -> anyhow::Result<()> {
440    let options = eframe::NativeOptions {
441        viewport: egui::ViewportBuilder::default().with_inner_size([1280.0, 800.0]),
442        renderer: eframe::Renderer::Wgpu,
443        ..Default::default()
444    };
445    eframe::run_native(
446        "Ranim Inspector",
447        options,
448        Box::new(move |_| Ok(Box::new(InspectorApp::new(app)))),
449    )
450    .map_err(|error| anyhow::anyhow!("failed to launch scene inspector: {error}"))
451}

453#[cfg(test)]
454mod tests {
455    use super::*;
456    use crate::{App, RanimScene};

470    #[test]
471    fn browsing_is_lazy_and_preserves_non_send_inputs() {
472        struct Recipe;
473        impl AnimRecipe<Rc<f64>> for Recipe {
474            fn build(&self, input: &Rc<f64>) -> AnimNode {
475                AnimNode::from_eval(Static(**input), 1.0)
476            }
477        }
478        let builds = Arc::new(AtomicUsize::new(0));
479        let counter = Arc::clone(&builds);
480        let mut project = App::new();
481        project.add_scene(Scene::new("sample", move |r: &mut RanimScene| {
482            counter.fetch_add(1, Ordering::Relaxed);
483            r.play(Rc::new(1.0).anim(Recipe).with_duration(2.0));
484        }));
485        let mut app = InspectorApp::new(project.build());
486        assert_eq!(builds.load(Ordering::Relaxed), 0);
487        app.selected_scene = Some(0);
488        app.build_selected_scene();
489        app.build_selected_scene();
490        assert_eq!(builds.load(Ordering::Relaxed), 1);
491        let snapshot = app.snapshots[&0].as_ref().unwrap();
492        let roots = snapshot.scene.root.definitions();
493        assert_eq!(snapshot.scene.root.duration_secs(), 2.0);
494        let AnimDefNodeContent::Leaf(leaf) = &node_at_path(roots, &[0]).unwrap().content else {
495            panic!()
496        };
497        assert_eq!(
498            *leaf
499                .downcast_ref::<AnimDef<Rc<f64>, Recipe>>()
500                .unwrap()
501                .initial,
502            1.0
503        );
504        assert!(node_at_path(roots, &[0, 0]).is_none());
505        assert!(node_at_path(roots, &[]).is_none());
506    }
