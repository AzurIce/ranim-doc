1//! User-facing composition helpers built on the core node language.
2//!
3//! `AnimSequence`, `AnimStack`, and `AnimLagged` are sugar/elaboration
4//! constructors: they own their child definitions and lower to the closed
5//! [`crate::animation::node::NodeContent`] vocabulary in
6//! [`crate::animation::def_node::AnimDefNode::build`]. The primitives they
7//! lower to stay in core so every interpreter (visual eval, audio bake, preview
8//! info) can see the same structure.
9//!
10pub mod lagged;
11pub mod sequence;
12pub mod stack;

14use crate::animation::compose::{lagged::AnimLagged, sequence::AnimSequence, stack::AnimStack};
15use crate::animation::{build::Unplaced, def_node::AnimDefNode};

17/// Collect iterators of animations into containers.
18pub trait AnimIterExt: Iterator + Sized {
19    /// Collect the animations into an [`AnimStack`] (all at the same origin).
20    fn into_stack(self) -> AnimStack
21    where
22        Self::Item: Into<AnimDefNode>,
23    {
24        self.collect()
25    }
26
27    /// Collect the animations into an [`AnimSequence`] (played in order).
28    fn into_seq(self) -> AnimSequence
29    where
30        Self::Item: Unplaced + 'static,
31    {
32        self.collect()
33    }
34
35    /// Collect the animations into an [`AnimLagged`] with the given stagger ratio.
36    fn into_lagged(self, lag_ratio: f64) -> AnimLagged
37    where
38        Self::Item: Unplaced + 'static,
39    {
40        let mut lagged = AnimLagged::new(lag_ratio);
41        for animation in self {
42            lagged.push(animation);
43        }
44        lagged
45    }
46}
47
48impl<I: Iterator> AnimIterExt for I {}

50#[cfg(test)]
51mod tests {
52    use super::*;
53    use crate::{
54        animation::{
55            AnimRecipe,
56            def_node::{AnimDef, AnimDefNodeContent},
57            eval::{OutputContext, Static},
58            node::AnimNode,
59        },
60        frame::EvaluatedFrame,
61    };
62
63    #[test]
64    fn holds_and_lagged_fills_keep_captured_snapshots_when_sources_are_edited() {
65        struct Recipe;
66        impl AnimRecipe<f64> for Recipe {
67            fn build(&self, initial: &f64) -> AnimNode {
68                AnimNode::from_eval(Static(*initial), 2.0)
69            }
70        }
71        fn edit_sources(node: &mut AnimDefNode) {
72            match &mut node.content {
73                AnimDefNodeContent::Children(children) => {
74                    children.iter_mut().for_each(edit_sources)
75                }
76                AnimDefNodeContent::Leaf(leaf) => {
77                    if let Some(definition) = leaf.downcast_mut::<AnimDef<f64, Recipe>>() {
78                        definition.initial += 10.0;
79                    }
80                }
81            }
82        }
83        fn values(node: &AnimNode, sec: f64) -> Vec<f64> {
84            let mut frame = EvaluatedFrame::new();
85            node.eval_at(sec, &mut OutputContext::new(&mut frame, 0));
86            frame.items::<f64>().iter().map(|item| item.value).collect()
87        }
88
89        let mut sequence = AnimSequence::new();
90        sequence.push(AnimDef::new(1.0, Recipe)).hold(1.0);
91        edit_sources(&mut sequence.animations[0]);
92        let mut lagged = AnimLagged::new(0.5);
93        lagged
94            .push(AnimDef::new(2.0, Recipe))
95            .push(AnimDef::new(3.0, Recipe));
96        let mut lagged: AnimDefNode = lagged.into();
97        edit_sources(&mut lagged);
98        let definition: AnimDefNode = crate::stack![sequence, lagged.at(5.0)].into();
99
100        for runtime in [definition.build(), definition.build()] {
101            assert_eq!(values(&runtime, 1.0), vec![11.0]);
102            assert_eq!(values(&runtime, 2.5), vec![1.0]);
103            assert_eq!(values(&runtime, 5.5), vec![12.0, 3.0]);
104            assert_eq!(values(&runtime, 7.5), vec![2.0, 13.0]);
105        }
106    }
