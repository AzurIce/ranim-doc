12/// How an [`AnimLagged`] fills the time outside a child's window.
13///
14/// Fills are materialized as real static cells at `build` time (sampled from
15/// the child's window edges), so the preview timeline shows exactly what is
16/// rendered — there is no hidden clamping rule.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum LaggedFill {
19    /// Render nothing outside the window.
20    Empty,
21    /// Keep showing the window-edge state with a static animation
22    /// (the initial state before, the final state after).
23    Hold,
24}
25
26/// Dynamic lagged (staggered, end-filled) animation container.
27///
28/// Children are pushed un-placed ([`Placeable`]); the container computes the
29/// placement itself: child `i` starts at `start_{i-1} + lag_ratio · d_{i-1}`.
30/// `lag_ratio` interpolates between the other two containers:
31///
32/// - `0.0` — all children start together (like [`AnimStack`](super::stack::AnimStack));
33/// - `1.0` — each child starts when the previous ends (like [`AnimSequence`]);
34/// - in between — overlapping succession.
35///
36/// By default the time outside a child's window is **filled with real static
37/// cells** ([`LaggedFill::Hold`] on both ends): each item is materialized at
38/// build time as a per-item sequence track `[leading fill][anim][trailing
39/// fill]` spanning the whole extent — before its start the item shows its
40/// initial state, after its end it keeps showing its final state, and the
41/// preview timeline shows exactly what is rendered. Configure with
42/// [`with_leading`](Self::with_leading) /
43/// [`with_trailing`](Self::with_trailing) — e.g. `Empty` leading makes items
44/// appear only at their window; to hide an item after its window instead of
45/// holding it, give it an animation that ends hidden, e.g.
46/// `seq![item.fade_in(), item.hide()]`.
47///
48/// Fills are sampled at build time: empty fills are skipped, and a
49/// zero-duration child gets no leading fill (it appears at its point, which is
50/// what "show from that point on" means). Because fills are build-time
51/// samples, children are expected to be pure (closed-form) animations — a
52/// stateful child's trailing fill would be its initial state, not its true
53/// final state.
54pub struct AnimLagged {
55    animations: Vec<AnimationCell>,
56    lag_ratio: f64,
57    /// Start offset for the next pushed child.
58    cursor_sec: f64,
59    duration_secs: f64,
60    leading: LaggedFill,
61    trailing: LaggedFill,
62}
63
64impl AnimLagged {
65    /// Create an empty lagged container with the given stagger ratio.
66    pub fn new(lag_ratio: f64) -> Self {
67        assert!(
68            lag_ratio.is_finite() && lag_ratio >= 0.0,
69            "lag ratio must be finite and non-negative"
70        );
71        Self {
72            animations: Vec::new(),
73            lag_ratio,
74            cursor_sec: 0.0,
75            duration_secs: 0.0,
76            leading: LaggedFill::Hold,
77            trailing: LaggedFill::Hold,
78        }
79    }
80
81    /// The stagger ratio between successive children.
82    pub fn lag_ratio(&self) -> f64 {
83        self.lag_ratio
84    }
85
86    /// Configure the fill before each child's window (default
87    /// [`LaggedFill::Hold`]).
88    pub fn with_leading(mut self, behavior: LaggedFill) -> Self {
89        self.leading = behavior;
90        self
91    }
92
93    /// Configure the fill after each child's window (default
94    /// [`LaggedFill::Hold`]).
95    pub fn with_trailing(mut self, behavior: LaggedFill) -> Self {
96        self.trailing = behavior;
97        self
98    }
99
100    /// Add an animation, placed by the container's stagger rule.
101    pub fn push<A: Placeable + 'static>(&mut self, animation: A) -> &mut Self {
102        let mut animation = animation.build();
103        let duration_secs = animation.duration_secs();
104        animation.shift_by(self.cursor_sec);
105        self.duration_secs = self.duration_secs.max(animation.time_range.end);
106        self.animations.push(animation);
107        self.cursor_sec += self.lag_ratio * duration_secs;
108        self
109    }
110
111    /// Materialize each child as a per-item sequence track:
112    /// `[leading fill][anim][trailing fill]` (empty fills skipped).
113    ///
114    /// The lagged container is thus a stack of per-item sequences — each
115    /// item's track spans the whole extent, with its window-edge states held
116    /// by real static cells.
117    fn materialize_fills(&mut self) {
118        let total = self.duration_secs;
119        let children = std::mem::take(&mut self.animations);
120        let mut animations = Vec::with_capacity(children.len());
121        for child in children {
122            let start = child.time_range.start;
123            let end = child.time_range.end;
124            let mut track = AnimSequence::new();
125            if self.leading == LaggedFill::Hold && start > 0.0 && child.duration_secs() > 0.0 {
126                let mut state = Vec::new();
127                child.eval_at(start, &mut state);
128                if !state.is_empty() {
129                    track.animations.push(static_cell(state, 0.0..start));
130                }
131            }
132            track.animations.push(child);
133            if self.trailing == LaggedFill::Hold && end < total {
134                let child = track.animations.last().unwrap();
135                let mut state = Vec::new();
136                child.eval_at(end, &mut state);
137                if !state.is_empty() {
138                    track.animations.push(static_cell(state, end..total));
139                }
140            }
141            track.cursor_sec = total;
142            animations.push(track.build());
143        }
144        self.animations = animations;
145    }
146
147    /// Current total extent (the last child's end).
148    pub fn duration_secs(&self) -> f64 {
149        self.duration_secs
150    }
151
152    /// Borrow the direct child animations in local container coordinates.
153    pub fn built_animations(&self) -> &[AnimationCell] {
154        &self.animations
155    }
156
157    /// Consume this container into its direct child animations.
158    pub fn into_built_animations(self) -> Vec<AnimationCell> {
159        self.animations
160    }
161}

163impl Placeable for AnimLagged {}
164impl Animation for AnimLagged {
165    fn build(mut self) -> AnimationCell {
166        self.materialize_fills();
167        let duration_secs = self.duration_secs;
168        AnimationCell {
169            inner: Box::new(self),
170            rate_func: linear,
171            time_range: 0.0..duration_secs,
172            enabled: true,
173            anim_name: type_name::<Self>(),
174        }
175    }
176}

178impl EvalDyn for AnimLagged {
179    fn eval_dyn(&self, alpha: f64, output: &mut Vec<DynItem>) {
180        let content_sec = self.duration_secs * alpha;
181        for animation in &self.animations {
182            if animation.contains_sec(content_sec, self.duration_secs) {
183                animation.eval_at(content_sec, output);
184            }
185        }
186    }
187
188    fn info_kind(&self) -> AnimationInfoKind {
189        AnimationInfoKind::Lagged
190    }
191
192    fn content_duration_secs(&self) -> f64 {
193        self.duration_secs
194    }
195
196    fn child_infos(&self) -> Vec<AnimationInfo> {
197        self.animations
198            .iter()
199            .map(AnimationCell::animation_info)
200            .collect()
201    }
202}

204/// Construct an [`AnimLagged`] with a stagger ratio, pushing each animation in order.
205#[macro_export]
206macro_rules! lagged {
207    ($lag_ratio:expr; $($animation:expr),* $(,)?) => {
208        {
209            #[allow(unused_mut)]
210            let mut lagged = $crate::animation::lagged::AnimLagged::new($lag_ratio);
211            $(lagged.push($animation);)*
212            lagged
213        }
214    };
