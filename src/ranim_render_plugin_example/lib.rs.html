1//! A project with a private animation output type and its own render setup.
2
3use ranim::{
4    AppExt, Plugin, RanimScene, Scene,
5    color::palette::css,
6    items::{
7        RanimItemsPlugin,
8        camera::CameraFrame,
9        mesh::MeshItem,
10        vitem::{VItem, geometry::Square},
11    },
12    prelude::{FillColor, PlaybackExt, ShiftTransform, Static, StrokeColor, StrokeWidth},
13    render::{
14        RenderSetup,
15        extract::{Extract, ExtractContext, ExtractPlugin},
16    },
17};
18
19/// The host cannot name this complete animation output type.
20#[derive(Clone)]
21struct PrivateDrawing {
22    vector: VItem,
23    mesh: MeshItem,
24}
25
26impl Extract for PrivateDrawing {
27    fn extract(&self, ctx: &mut ExtractContext<'_>) {
28        self.vector.extract(ctx);
29        self.mesh.extract(ctx);
30    }
31}
32
33fn scene(scene: &mut RanimScene) {
34    use ranim::glam::DVec3;
35
36    let mut vector = VItem::from(Square::new(1.6));
37    vector
38        .shift(DVec3::new(-1.5, 0.0, 0.0))
39        .set_fill_color(css::RED)
40        .set_fill_opacity(1.0)
41        .set_stroke_color(css::RED)
42        .set_stroke_width(0.04);
43    let mesh = MeshItem::from_indexed_vertices(
44        vec![
45            DVec3::new(0.6, -0.8, 0.0),
46            DVec3::new(2.4, -0.8, 0.0),
47            DVec3::new(1.5, 0.9, 0.0),
48        ],
49        vec![0, 1, 2],
50    )
51    .with_color(css::BLUE);
52
53    scene.play(Static(CameraFrame::default()).with_duration(1.0));
54    scene.play(Static(PrivateDrawing { vector, mesh }).with_duration(1.0));
55}
56
57/// Configure the project's scene and rendering support.
58pub struct ProjectPlugin;
59
60impl Plugin for ProjectPlugin {
61    fn build(&self, app: &mut bevy_app::App) {
62        app.insert_resource(RenderSetup::new(|render_app| {
63            render_app.add_plugins((RanimItemsPlugin, ExtractPlugin::<PrivateDrawing>::default()));
64        }))
65        .add_scene(Scene::new("render_plugin", scene));
66    }
67}
