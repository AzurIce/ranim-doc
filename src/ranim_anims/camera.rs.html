1//! Camera orbit parameters and their prepared execution data.
2
3use ranim_core::{
4    animation::{AnimRecipe, AnimWriteBack, eval::Eval, node::AnimNode},
5    glam::DVec3,
6};
7use ranim_items::camera::CameraFrame;
8
9/// Orbit around a target in the XY plane (Z-up).
10#[derive(Clone, Copy, Debug)]
11pub struct Orbit {
12    /// Orbit target.
13    pub target: DVec3,
14    /// Total rotation, in radians.
15    pub angle: f64,
16}
17
18impl Orbit {
19    /// Describe an orbit around `target` through `angle` radians.
20    pub fn new(target: DVec3, angle: f64) -> Self {
21        Self { target, angle }
22    }
23}
24
25/// Prepared camera orbit with derived spherical coordinates.
26pub struct OrbitEval {
27    src: CameraFrame,
28    target: DVec3,
29    distance: f64,
30    phi: f64,
31    theta0: f64,
32    angle: f64,
33}
34
35impl Orbit {
36    fn prepare(&self, input: &CameraFrame) -> OrbitEval {
37        let offset = input.pos - self.target;
38        let distance = offset.length();
39        let phi = if distance > 0.0 {
40            (offset.z / distance).acos()
41        } else {
42            0.0
43        };
44        OrbitEval {
45            src: input.clone(),
46            target: self.target,
47            distance,
48            phi,
49            theta0: offset.y.atan2(offset.x),
50            angle: self.angle,
51        }
52    }
53}
54
55impl AnimRecipe<CameraFrame> for Orbit {
56    fn build(&self, input: &CameraFrame) -> AnimNode {
57        AnimNode::from_eval(self.prepare(input), 1.0)
58    }
59}
60
61impl AnimWriteBack<CameraFrame> for Orbit {
62    fn end_value(&self, input: &CameraFrame) -> CameraFrame {
63        self.prepare(input).eval_alpha(1.0)
64    }
65}
66
67impl Eval for OrbitEval {
68    type Output = CameraFrame;
69    fn eval_alpha(&self, alpha: f64) -> CameraFrame {
70        let mut result = self.src.clone();
71        result.set_spherical(
72            self.phi,
73            self.theta0 + self.angle * alpha,
74            self.distance,
75            self.target,
76        );
77        result
78    }
79}
