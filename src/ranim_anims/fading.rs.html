1//! Opacity animation parameters and prepared evaluators.
2
3use ranim_core::{
4    animation::{AnimRecipe, AnimWriteBack, eval::Eval, node::AnimNode},
5    traits::{Interpolatable, Opacity},
6};
7
8/// Item capabilities needed for opacity interpolation.
9pub trait FadingRequirement: Opacity + Interpolatable + Clone {}
10impl<T: Opacity + Interpolatable + Clone> FadingRequirement for T {}
11
12/// Fade from transparent to the input item's original opacity.
13#[derive(Clone, Copy, Debug, Default)]
14pub struct FadeIn;
15
16/// Fade from the input item's opacity to transparent.
17#[derive(Clone, Copy, Debug, Default)]
18pub struct FadeOut;
19
20/// Prepared opacity interpolation; derived states are not construction parameters.
21pub struct FadeEval<T> {
22    src: T,
23    dst: T,
24}
25
26impl<T: FadingRequirement + Send + Sync + 'static> AnimRecipe<T> for FadeIn {
27    fn build(&self, input: &T) -> AnimNode {
28        let mut src = input.clone();
29        src.set_opacity(0.0);
30        AnimNode::from_eval(
31            FadeEval {
32                src,
33                dst: input.clone(),
34            },
35            1.0,
36        )
37    }
38}
39impl<T: FadingRequirement + Send + Sync + 'static> AnimWriteBack<T> for FadeIn {
40    fn end_value(&self, input: &T) -> T {
41        input.clone()
42    }
43}
44
45impl<T: FadingRequirement + Send + Sync + 'static> AnimRecipe<T> for FadeOut {
46    fn build(&self, input: &T) -> AnimNode {
47        let mut dst = input.clone();
48        dst.set_opacity(0.0);
49        AnimNode::from_eval(
50            FadeEval {
51                src: input.clone(),
52                dst,
53            },
54            1.0,
55        )
56    }
57}
58impl<T: FadingRequirement + Send + Sync + 'static> AnimWriteBack<T> for FadeOut {
59    fn end_value(&self, input: &T) -> T {
60        let mut result = input.clone();
61        result.set_opacity(0.0);
62        result
63    }
64}
65
66impl<T: FadingRequirement> Eval for FadeEval<T> {
67    type Output = T;
68    fn eval_alpha(&self, alpha: f64) -> T {
69        self.src.lerp(&self.dst, alpha)
70    }
71}
