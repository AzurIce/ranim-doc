25/// Output format for `ranim inspect` subcommands.
26#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, ValueEnum)]
27pub enum FormatArg {
28    /// Human-readable text output.
29    #[default]
30    Text,
31    /// JSON output.
32    Json,
33}
34
35#[derive(Debug, Subcommand)]
36pub enum InspectCommand {
37    /// List available scenes and their output summaries without constructing scenes.
38    Scenes {
39        #[arg(long, value_enum, default_value_t = FormatArg::Text)]
40        format: FormatArg,
41    },
42    /// Build a scene and print its hierarchical animation tree.
43    Tree {
44        /// Scene name. Optional when the library contains exactly one scene.
45        scene: Option<String>,
46        #[arg(long, value_enum, default_value_t = FormatArg::Text)]
47        format: FormatArg,
48    },
49    /// Sample a scene at a time and print the evaluated frame items.
50    Frame {
51        /// Scene name.
52        scene: String,
53        /// Sample time in seconds.
54        #[arg(long)]
55        at: f64,
56        #[arg(long, value_enum, default_value_t = FormatArg::Text)]
57        format: FormatArg,
58        /// Include full geometry data (points, indices, colors, normals).
59        #[arg(long)]
60        verbose: bool,
61    },
62}
63
64pub fn inspect_command(args: &CliArgs, command: InspectCommand) -> Result<()> {
65    let lib = load_user_library(args)?;
66    match command {
67        InspectCommand::Scenes { format } => scenes_command(&lib, format),
68        InspectCommand::Tree { scene, format } => tree_command(&lib, scene.as_deref(), format),
69        InspectCommand::Frame {
70            scene,
71            at,
72            format,
73            verbose,
74        } => frame_command(&lib, &scene, at, format, verbose),
75    }
76}

106fn scenes_output(lib: &RanimUserLibrary) -> ScenesOutput {
107    let scenes = lib
108        .scenes()
109        .map(|scene| SceneDto {
110            name: scene.name,
111            clear_color: scene.config.clear_color,
112            outputs: scene
113                .outputs
114                .iter()
115                .map(|output| OutputDto {
116                    width: output.width,
117                    height: output.height,
118                    fps: output.fps,
119                    format: output.format.to_string(),
120                    dir: output.dir.clone(),
121                    name_template: output
122                        .name_template
123                        .clone()
124                        .unwrap_or_else(|| DEFAULT_NAME_TEMPLATE.to_string()),
125                    save_frames: output.save_frames,
126                })
127                .collect(),
128        })
129        .collect();
130    ScenesOutput {
131        schema_version: 1,
132        scenes,
133    }
134}

136fn scenes_command(lib: &RanimUserLibrary, format: FormatArg) -> Result<()> {
137    let output = scenes_output(lib);
138    match format {
139        FormatArg::Json => print_json(&output),
140        FormatArg::Text => {
141            for scene in &output.scenes {
142                println!("scene {}", scene.name);
143                println!("  clear_color: {}", scene.clear_color);
144                println!("  outputs:");
145                if scene.outputs.is_empty() {
146                    println!("    (none)");
147                }
148                for output in &scene.outputs {
149                    println!(
150                        "    - {}x{} @{}fps {} -> {} save_frames={} name_template=\"{}\"",
151                        output.width,
152                        output.height,
153                        output.fps,
154                        output.format,
155                        output.dir,
156                        output.save_frames,
157                        output.name_template
158                    );
159                }
160            }
161            Ok(())
162        }
163    }

219fn animation_info_to_tree_node(info: &AnimationInfo, path: Vec<usize>) -> TreeNodeDto {
220    TreeNodeDto {
221        path: path.clone(),
222        kind: animation_kind_str(info.kind).to_string(),
223        anim_name: info.anim_name.clone(),

225        content_duration_secs: info.content_duration_secs,
226        rate_func: rate_func_name(info.rate_func).to_string(),
227        enabled: info.enabled,
228        sim_step: info.sim_step,
229        children: info

231            .iter()

233            .map(|(idx, child)| {

235                child_path.push(idx);

256fn rate_func_name(rate_func: fn(f64) -> f64) -> &'static str {
257    // Function pointers are opaque in Rust and addresses are not guaranteed to
258    // be comparable across codegen units, so identify the known rate functions
259    // by their values on a small probe set instead.
260    const PROBE_POINTS: [f64; 6] = [0.0, 0.1, 0.25, 0.5, 0.75, 1.0];
261    type KnownRateFunc = (&'static str, fn(f64) -> f64);
262    const KNOWN: &[KnownRateFunc] = &[
263        ("linear", rate_functions::linear),
264        ("smooth", rate_functions::smooth),
265        ("ease_in_quad", rate_functions::ease_in_quad),
266        ("ease_out_quad", rate_functions::ease_out_quad),
267        ("ease_in_out_quad", rate_functions::ease_in_out_quad),
268        ("ease_in_cubic", rate_functions::ease_in_cubic),
269        ("ease_out_cubic", rate_functions::ease_out_cubic),
270        ("ease_in_out_cubic", rate_functions::ease_in_out_cubic),
271    ];
272
273    KNOWN
274        .iter()
275        .find(|(_, known)| {
276            PROBE_POINTS
277                .iter()
278                .all(|&point| (known(point) - rate_func(point)).abs() < 1e-12)
279        })
280        .map(|(name, _)| *name)
281        .unwrap_or("custom")
282}

284fn short_anim_name(anim_name: &str) -> &str {
285    let without_generics = anim_name.split('<').next().unwrap_or(anim_name);
286    without_generics
287        .rsplit("::")
288        .next()
289        .unwrap_or(without_generics)
290}

292fn tree_command(lib: &RanimUserLibrary, scene_name: Option<&str>, format: FormatArg) -> Result<()> {
293    let scene = select_scene(lib, scene_name)?;
294    let sealed = scene.constructor.build_scene();
295    let time_marks = sealed
296        .time_marks()
297        .iter()
298        .map(|(sec, mark)| TimeMarkDto {
299            sec: *sec,
300            kind: match mark {
301                ranim::core::TimeMark::Capture(_) => "capture".to_string(),
302            },
303            label: match mark {
304                ranim::core::TimeMark::Capture(label) => label.clone(),
305            },
306        })
307        .collect();
308    let output = tree_output(
309        scene.name.clone(),
310        sealed.total_secs(),
311        time_marks,
312        sealed.get_animation_infos(),
313    );
314    match format {
315        FormatArg::Json => print_json(&output),
316        FormatArg::Text => {
317            println!("scene {}", output.scene);
318            println!("total_secs: {}", output.total_secs);
319            println!("time_marks:");
320            for mark in &output.time_marks {
321                println!("  - {} {} {:?}", mark.sec, mark.kind, mark.label);
322            }
323            println!("tree:");
324            for node in &output.tree {
325                print_tree_node(node, 0);
326            }
327            Ok(())
328        }
329    }
330}
331
332fn print_tree_node(node: &TreeNodeDto, depth: usize) {
333    let path = node
334        .path
335        .iter()
336        .map(|idx| idx.to_string())
337        .collect::<Vec<_>>()
338        .join(", ");
339    let sim_step = node
340        .sim_step
341        .map(|step| format!(" sim_step={step}"))
342        .unwrap_or_default();
343    println!(
344        "{}[{}] {} {} [{}..{}] content={} rate_func={} enabled={}{}",
345        "  ".repeat(depth),
346        path,
347        short_anim_name(&node.anim_name),
348        node.kind,
349        node.range[0],
350        node.range[1],
351        node.content_duration_secs,
352        node.rate_func,
353        node.enabled,
354        sim_step
355    );
356    for child in &node.children {
357        print_tree_node(child, depth + 1);
358    }
359}

448fn frame_command(
449    lib: &RanimUserLibrary,
450    scene_name: &str,
451    at: f64,
452    format: FormatArg,
453    verbose: bool,
454) -> Result<()> {
455    let scene = select_scene(lib, Some(scene_name))?;
456    let sealed = scene.constructor.build_scene();
457    let total_secs = sealed.total_secs();
458    let mut evaluator = sealed.into_evaluator(120.0);
459    let infos = evaluator.animation_infos();
460    let mut frame = EvaluatedFrame::new();
461    evaluator.sample_at(at, &mut frame);
462
463    let items = frame
464        .iter()
465        .enumerate()
466        .map(|(z_order, ((animation_id, part), item))| {
467            FrameItemDto::from_core_item(*animation_id, *part, item, &infos, z_order, verbose)
468        })
469        .collect();
470
471    let output = FrameOutput {
472        schema_version: 1,
473        scene: scene.name.clone(),
474        at,
475        total_secs,
476        items,
477    };
478
479    match format {
480        FormatArg::Json => print_json(&output),
481        FormatArg::Text => {
482            println!(
483                "scene {} @{} total_secs={}",
484                output.scene, output.at, output.total_secs
485            );
486            for item in &output.items {
487                print_frame_item(item);
488            }
489            Ok(())
490        }
491    }
492}

494impl FrameItemDto {
495    fn from_core_item(
496        animation_id: usize,
497        part: usize,
498        item: &CoreItem,
499        infos: &[AnimationInfo],
500        z_order: usize,
501        verbose: bool,
502    ) -> Self {
503        let (kind, data) = match item {
504            CoreItem::VItem(item) => ("vitem", FrameItemDataDto::from_vitem(item, verbose)),
505            CoreItem::MeshItem(item) => ("mesh", FrameItemDataDto::from_mesh(item, verbose)),
506            CoreItem::CameraFrame(item) => ("camera", FrameItemDataDto::from_camera(item)),
507        };
508
509        Self {
510            z_order,
511            id: [animation_id, part],
512            animation_id,
513            part,
514            kind: kind.to_string(),
515            source: source_from_infos(infos, animation_id),
516            data,
517        }
518    }
519}

521impl FrameItemDataDto {
522    fn from_vitem(item: &VItem, verbose: bool) -> Self {
523        // Core VItem data lives in the item's local space; the inspect dump
524        // reports world-space values, so apply the stored transform here
525        // (in the same f32 precision used by the renderer).
526        let normal_matrix = Mat3::from_mat4(item.transform).inverse().transpose();
527        let vpoints = VPointVec(
528            item.points
529                .iter()
530                .map(|point| item.transform.transform_point3(point.truncate()).as_dvec3())
531                .collect(),
532        );
533        let subpaths = if vpoints.is_empty() {
534            Vec::new()
535        } else {
536            vpoints.get_subpaths()
537        };
538        let closed = subpaths
539            .iter()
540            .map(|subpath| {
541                get_subpath_closed_flag(subpath)
542                    .map(|(_, closed)| closed)
543                    .unwrap_or(false)
544            })
545            .collect();
546        let normal_local = item
547            .normal
548            .unwrap_or_else(|| vitem_normal_from_points(&item.points));
549        let normal = (normal_matrix * normal_local).normalize_or_zero();
550        let bounds = dvec3_bounds(vpoints.aabb());
551
552        Self::VItem(VItemDataDto {
553            point_count: item.points.len(),
554            subpath_count: subpaths.len(),
555            closed,
556            fill_colors: item.fill_rgbas.iter().map(rgba_to_hex).collect(),
557            stroke_colors: item.stroke_rgbas.iter().map(rgba_to_hex).collect(),
558            stroke_widths: item.stroke_widths.iter().map(|width| width.0).collect(),
559            normal: normal.to_array(),
560            bounds,
561            points: verbose.then(|| {
562                item.points
563                    .iter()
564                    .map(|point| {
565                        let world = item.transform.transform_point3(point.truncate());
566                        [world.x, world.y, world.z, point.w]
567                    })
568                    .collect()
569            }),
570        })
571    }
572
573    fn from_mesh(item: &MeshItem, verbose: bool) -> Self {
574        let transformed_points = item
575            .points
576            .iter()
577            .map(|point| item.transform.transform_point3(*point))
578            .collect::<Vec<_>>();
579        let bounds = vec3_bounds(&transformed_points);
580
581        Self::Mesh(MeshDataDto {
582            point_count: item.points.len(),
583            triangle_count: item.triangle_indices.len() / 3,
584            transform: item.transform.to_cols_array(),
585            bounds,
586            points: verbose.then(|| {
587                transformed_points
588                    .iter()
589                    .map(|point| point.to_array())
590                    .collect()
591            }),
592            triangle_indices: verbose.then(|| item.triangle_indices.clone()),
593            vertex_colors: verbose.then(|| item.vertex_colors.iter().map(rgba_to_hex).collect()),
594            vertex_normals: verbose.then(|| {
595                item.vertex_normals
596                    .iter()
597                    .map(|normal| normal.to_array())
598                    .collect()
599            }),
600        })
601    }
602
603    fn from_camera(item: &CameraFrame) -> Self {
604        Self::Camera(CameraDataDto {
605            pos: dvec3_to_array(item.pos),
606            facing: dvec3_to_array(item.facing),
607            up: dvec3_to_array(item.up),
608            near: item.near,
609            far: item.far,
610            perspective_blend: item.perspective_blend,
611            frame_height: item.frame_height,
612            scale: item.scale,
613            fovy: item.fovy,
614        })
615    }
616}

635fn dvec3_bounds(bounds: [DVec3; 2]) -> BoundsDto {
636    BoundsDto {
637        min: dvec3_to_array(bounds[0]),
638        max: dvec3_to_array(bounds[1]),
639    }

642fn vec3_bounds(points: &[ranim::glam::Vec3]) -> BoundsDto {
643    if points.is_empty() {
644        return BoundsDto {
645            min: [0.0; 3],
646            max: [0.0; 3],
647        };
648    }
649    let mut min = points[0];
650    let mut max = points[0];
651    for point in &points[1..] {
652        min = min.min(*point);
653        max = max.max(*point);
654    }
655    BoundsDto {
656        min: [min.x as f64, min.y as f64, min.z as f64],
657        max: [max.x as f64, max.y as f64, max.z as f64],
658    }
659}

661fn dvec3_to_array(v: DVec3) -> [f64; 3] {
662    [v.x, v.y, v.z]
663}

665fn rgba_to_hex(rgba: &Rgba) -> String {
666    let srgb: AlphaColor<Srgb> = (*rgba).into();
667    let [r, g, b, a] = srgb.to_rgba8().to_u8_array();
668    format!("#{r:02x}{g:02x}{b:02x}{a:02x}")

671fn print_frame_item(item: &FrameItemDto) {
672    println!(
673        "[{}] id=[{}, {}] {} source=[{}] {} {} range=[{}..{}]",
674        item.z_order,
675        item.id[0],
676        item.id[1],
677        item.kind,
678        item.source
679            .path
680            .iter()
681            .map(|idx| idx.to_string())
682            .collect::<Vec<_>>()
683            .join(", "),
684        item.source.kind,
685        short_anim_name(&item.source.anim_name),
686        item.source.range[0],
687        item.source.range[1],
688    );
689    match &item.data {
690        FrameItemDataDto::VItem(data) => {
691            println!(
692                "  point_count={} subpath_count={} closed={:?}",
693                data.point_count, data.subpath_count, data.closed
694            );
695            println!(
696                "  fill={:?} stroke={:?} stroke_widths={:?}",
697                data.fill_colors, data.stroke_colors, data.stroke_widths
698            );
699            println!(
700                "  normal={:?} bounds=min:{:?} max:{:?}",
701                data.normal, data.bounds.min, data.bounds.max
702            );
703            if let Some(points) = &data.points {
704                println!("  points={points:?}");
705            }
706        }
707        FrameItemDataDto::Mesh(data) => {
708            println!(
709                "  point_count={} triangle_count={} bounds=min:{:?} max:{:?}",
710                data.point_count, data.triangle_count, data.bounds.min, data.bounds.max
711            );
712            if let Some(points) = &data.points {
713                println!("  points={points:?}");
714            }
715            if let Some(triangle_indices) = &data.triangle_indices {
716                println!("  triangle_indices={triangle_indices:?}");
717            }
718            if let Some(vertex_colors) = &data.vertex_colors {
