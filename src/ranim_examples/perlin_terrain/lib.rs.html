30fn perm(i: usize) -> usize {
31    P[i & 255]
32}

34fn fade(t: f64) -> f64 {
35    t * t * t * (t * (t * 6.0 - 15.0) + 10.0)
36}
37
38fn fade_d(t: f64) -> f64 {
39    30.0 * t * t * (t * (t - 2.0) + 1.0)
40}

42fn grad(hash: usize, x: f64, y: f64) -> f64 {
43    const SQRT5_INV: f64 = 0.4472135954999579; // 1/sqrt(5)
44    const TABLE: [(f64, f64); 8] = [
45        (1.0, 2.0),
46        (1.0, -2.0),
47        (-1.0, 2.0),
48        (-1.0, -2.0),
49        (2.0, 1.0),
50        (2.0, -1.0),
51        (-2.0, 1.0),
52        (-2.0, -1.0),
53    ];
54    let (gx, gy) = TABLE[hash & 7];
55    (x * gx + y * gy) * SQRT5_INV
56}

58fn noise(x: f64, y: f64) -> f64 {
59    let xi = x.floor() as i64 as usize;
60    let yi = y.floor() as i64 as usize;
61
62    let xf = x - x.floor();
63    let yf = y - y.floor();
64
65    let u = fade(xf);
66    let v = fade(yf);
67
68    let a = perm(xi) + (yi & 255);
69    let b = perm(xi.wrapping_add(1)) + (yi & 255);
70    let p0 = grad(perm(a), xf, yf);
71    let p1 = grad(perm(b), xf - 1.0, yf);
72    let p2 = grad(perm(a + 1), xf, yf - 1.0);
73    let p3 = grad(perm(b + 1), xf - 1.0, yf - 1.0);
74
75    fn lerp(t: f64, a: f64, b: f64) -> f64 {
76        a + t * (b - a)
77    }
78
79    lerp(v, lerp(u, p0, p1), lerp(u, p2, p3))
80}

82fn noise_with_derivative(x: f64, y: f64) -> (f64, f64, f64) {
83    let xi = x.floor() as i64 as usize;
84    let yi = y.floor() as i64 as usize;
85
86    let xf = x - x.floor();
87    let yf = y - y.floor();
88
89    let u = fade(xf);
90    let v = fade(yf);
91
92    let a = perm(xi) + (yi & 255);
93    let b = perm(xi.wrapping_add(1)) + (yi & 255);
94    let p0 = grad(perm(a), xf, yf);
95    let p1 = grad(perm(b), xf - 1.0, yf);
96    let p2 = grad(perm(a + 1), xf, yf - 1.0);
97    let p3 = grad(perm(b + 1), xf - 1.0, yf - 1.0);
98
99    fn lerp(t: f64, a: f64, b: f64) -> f64 {
100        a + t * (b - a)
101    }
102
103    let n = lerp(v, lerp(u, p0, p1), lerp(u, p2, p3));
104    let dndx = ((p1 - p0) + (p3 - p2 - p1 + p0) * v) * fade_d(xf);
105    let dndy = ((p2 - p0) + (p3 - p2 - p1 + p0) * u) * fade_d(yf);
106
107    (n, dndx, dndy)
108}

110fn fractal_noise(x: f64, y: f64, octaves: u32, persistence: f64) -> f64 {
111    let mut total = 0.0;
112    let mut frequency = 1.0;
113    let mut amplitude = 1.0;
114    let mut max_value = 0.0;
115
116    for _ in 0..octaves {
117        total += noise(x * frequency, y * frequency) * amplitude;
118        max_value += amplitude;
119        amplitude *= persistence;
120        frequency *= 2.0;
121    }
122
123    total / max_value
124}

126fn fractal_with_derivative_noise(x: f64, y: f64, octaves: u32, persistence: f64) -> f64 {
127    let mut total = 0.0;
128    let mut frequency = 1.0;
129    let mut amplitude = 1.0;
130    let mut max_value = 0.0;
131    let mut dx = 0.0;
132    let mut dy = 0.0;
133
134    for _ in 0..octaves {
135        let (n, dndx, dndy) = noise_with_derivative(x * frequency, y * frequency);
136        dx += dndx * amplitude;
137        dy += dndy * amplitude;
138        let factor = n / (1.0 + dx * dx + dy * dy);
139        total += amplitude * factor;
140        max_value += amplitude;
141        amplitude *= persistence;
142        frequency *= 2.0;
143    }
144
145    total / max_value
146}

150const GRID_SIZE: f64 = 32.0;
151const LATTICE_CNT: f64 = 8.0;
152const DEPTH: f64 = 3.0;
153const RESOLUTION: u32 = 512;

155const U_MIN: f64 = 0.0;
156const U_MAX: f64 = 30.0;
157const CENTER_OFFSET: f64 = 16.0;

159// Height-based color scale matching the Python reference.
160// Python: [(c, x * depth) for c, x in [...]]
161// Colors map to the Z-coordinate of the vertex (raw height, no scaling).
162#[allow(clippy::neg_multiply)]
163fn terrain_colorscale() -> Vec<(ranim::color::AlphaColor<ranim::color::Srgb>, f64)> {
164    vec![
165        (manim::BLUE_E, -1.0 * DEPTH),
166        (manim::BLUE_E, -0.9 * DEPTH),
167        (manim::BLUE_C, -0.8 * DEPTH),
168        (manim::GOLD_E, -0.7 * DEPTH),
169        (manim::GREY_BROWN, -0.1 * DEPTH),
170        (manim::GREY_BROWN, 0.1 * DEPTH),
171        (manim::GREY_D, 0.25 * DEPTH),
172        (manim::GREY_C, 0.6 * DEPTH),
173        (manim::WHITE, 0.7 * DEPTH),
174        (manim::WHITE, 1.0 * DEPTH),
175    ]
176}

180fn build_terrain_scene(r: &mut RanimScene, height_func: impl Fn(f64, f64) -> f64) {
181    let phi = 70.0 * PI / 180.0;
182    let theta = 30.0 * PI / 180.0;
183    let distance = 22.0;
184
185    let mut cam = CameraFrame::from_spherical(phi, theta, distance);
186    cam.fovy = 50.0 * PI / 180.0;
187
188    let colorscale = terrain_colorscale();
189
190    // Z-up: x = u - offset, y = v - offset, z = height
191    let terrain = Surface::from_uv_func(
192        |u, v| {
193            let x = u - CENTER_OFFSET;
194            let y = v - CENTER_OFFSET;
195            let z = height_func(u, v);
196            DVec3::new(x, y, z)
197        },
198        (U_MIN, U_MAX),
199        (U_MIN, U_MAX),
200        (RESOLUTION, RESOLUTION),
201    )
202    .with_fill_by_z(&colorscale);
203
204    let camera = cam
205        .orbit(DVec3::ZERO, 2.0)
206        .with_duration(5.0)
207        .with_rate_func(linear);
208    r.play(camera);
209    r.play(terrain.show().with_duration(5.0));
210}

214/// Basic Perlin noise terrain.
215#[scene]
216#[wasm_demo_doc]
217#[output(dir = "./output/perlin_terrain/perlin")]
218fn perlin(r: &mut RanimScene) {
219    // Python: get_noise(x, y, size) → noise(v, u) (note v, u swap)
220    // height = noise_func(v, u, size) * depth = noise * depth
221    build_terrain_scene(r, |u, v| {
222        let nx = v / GRID_SIZE * LATTICE_CNT;
223        let ny = u / GRID_SIZE * LATTICE_CNT;
224        noise(nx, ny) * DEPTH
225    });
226    r.insert_time_mark(0.0, TimeMark::Capture("preview-perlin.png".to_string()));
227}

229/// Fractal Perlin noise terrain (octave stacking).
230#[scene]
231#[wasm_demo_doc]
232#[output(dir = "./output/perlin_terrain/fractal")]
233fn fractal_perlin(r: &mut RanimScene) {
234    // Python: get_fractal_noise(x, y, size) * depth → fractal_noise(v, u) * depth
235    // height = noise_func(v, u, size) * depth = fractal_noise * depth * depth
236    build_terrain_scene(r, |u, v| {
237        let nx = v / GRID_SIZE * LATTICE_CNT;
238        let ny = u / GRID_SIZE * LATTICE_CNT;
239        fractal_noise(nx, ny, 8, 0.5) * DEPTH * DEPTH
240    });
241    r.insert_time_mark(
242        0.0,
243        TimeMark::Capture("preview-fractal-perlin.png".to_string()),
244    );
245}
