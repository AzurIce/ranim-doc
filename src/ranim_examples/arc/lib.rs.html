1use itertools::Itertools;
2use ranim::{
3    anims::fading::FadingAnim,
4    color,
5    color::HueDirection,
6    glam::dvec2,
7    items::vitem::{VItem, geometry::Arc},
8    prelude::*,
9    utils::rate_functions::smooth,
10};
11
12#[scene]
13#[wasm_demo_doc]
14#[output(dir = "./output/arc")]
15pub fn arc(r: &mut RanimScene) {
16    // let frame_size = app.camera().size;
17    let frame_size = dvec2(8.0 * 16.0 / 9.0, 8.0);
18    let frame_start = dvec2(frame_size.x / -2.0, frame_size.y / -2.0);
19
20    let start_color = color::color("#FF8080FF");
21    let end_color = color::color("#58C4DDFF");
22
23    let nrow = 10;
24    let ncol = 10;
25    let step_x = frame_size.x / ncol as f64;
26    let step_y = frame_size.y / nrow as f64;
27
28    let mut arcs = (0..nrow)
29        .cartesian_product(0..ncol)
30        .map(|(i, j)| {
31            let (i, j) = (i as f64, j as f64);
32
33            let angle = std::f64::consts::PI * (j + 1.0) / ncol as f64 * 360.0 / 180.0;
34            let radius = step_y / 2.0 * 0.8;
35            let color = start_color.lerp(
36                end_color,
37                i as f32 / (nrow - 1) as f32,
38                HueDirection::Increasing,
39            );
40            let offset = frame_start + dvec2(j * step_x + step_x / 2.0, i * step_y + step_y / 2.0);
41            let mut arc: VItem = Arc::new(angle, radius)
42                .with(|arc| {
43                    arc.stroke_width = 0.12 * (j as f32 + 0.02) / ncol as f32;
44                    arc.set_stroke_color(color);
45                })
46                .into();
47            arc.shift(offset.extend(0.0));
48            arc
49        })
50        .collect::<Vec<_>>();
51    let total_secs = 3.0;
52    let content = arcs
53        .iter_mut()
54        .map(|arc| arc.fade_in().with_rate_func(smooth))
55        .into_lagged(0.2)
56        .with_duration(total_secs);
57
58    r.play(CameraFrame::default().show().with_duration(total_secs));
59    r.play(content);
60
61    r.insert_time_mark(total_secs, TimeMark::Capture("preview.png".to_string()));
62}
