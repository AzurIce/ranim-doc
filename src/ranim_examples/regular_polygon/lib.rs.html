1use std::f64::consts::PI;
2
3use ranim::{
4    anims::morph::MorphAnim,
5    color::{HueDirection, palettes::manim},
6    glam::{DQuat, DVec3, dvec2},
7    items::vitem::geometry::RegularPolygon,
8    prelude::*,
9    utils::rate_functions::smooth,
10};
11
12#[scene]
13#[wasm_demo_doc]
14#[output(dir = "./output/regular_polygon")]
15pub fn regular_polygon(r: &mut RanimScene) {
16    let frame_size = dvec2(8.0 * 16.0 / 9.0, 8.0);
17    let max_radius = frame_size.x.max(frame_size.y) / 2.0 * 0.9;
18
19    let n_layers = 12;
20    let total_rotation = PI / 2.0; // 90° total rotation spread
21
22    // Colors
23    let start_color = manim::BLUE_C;
24    let end_color = manim::RED_C;
25
26    // Target radii
27    let radii: Vec<f64> = (0..n_layers)
28        .map(|i| max_radius * (i + 1) as f64 / n_layers as f64)
29        .collect();
30
31    // Create polygons starting from zero radius
32    let mut polygons = (0..n_layers)
33        .map(|i| {
34            let sides = i + 3;
35            let t = i as f32 / (n_layers - 1).max(1) as f32;
36            let color = start_color.lerp(end_color, t, HueDirection::Increasing);
37
38            RegularPolygon::new(sides, 0.0)
39                .with(|p| {
40                    p.set_stroke_color(color);
41                    p.stroke_width = 0.05;
42                    p.set_fill_color(color.with_alpha(0.08));
43                })
44                .transformed(Similarity::IDENTITY)
45        })
46        .collect::<Vec<_>>();
47
48    let mut content = AnimSequence::new();
49
50    // Phase 1: Expand from center with lagged effect
51    let expand = polygons
52        .iter_mut()
53        .zip(radii.iter())
54        .map(|(poly, &target_radius)| {
55            poly.morph(|p| {
56                p.inner.radius = target_radius;
57            })
58            .with_rate_func(smooth)
59        })
60        .into_lagged(0.2);
61    content.push(expand.with_duration(1.0));
62
63    // Phase 2: Rotate each layer at different speeds
64    let rotations: Vec<f64> = (0..n_layers)
65        .map(|i| total_rotation * (n_layers - i) as f64 / n_layers as f64)
66        .collect();
67
68    let rotate = polygons
69        .iter_mut()
70        .zip(rotations.iter())
71        .map(|(poly, &rot)| {
72            poly.morph(|p| {
73                p.transform = Similarity {
74                    scale: 1.0,
75                    rotation: DQuat::from_rotation_z(rot),
76                    translation: DVec3::ZERO,
77                };
78            })
79            .with_rate_func(smooth)
80        })
81        .into_lagged(0.2);
82    content.push(rotate.with_duration(1.0));
83
84    r.insert_time_mark(
85        content.cursor_sec(),
86        TimeMark::Capture("preview.png".to_string()),
87    );
88
89    content.hold(0.5);
90
91    // Phase 3: Collapse back to center (reverse order - outer first)
92    let collapse = polygons
93        .iter_mut()
94        .rev()
95        .map(|poly| {
96            poly.morph(|p| {
97                p.inner.radius = 0.0;
98            })
99            .with_rate_func(smooth)
100        })
101        .into_lagged(0.2);
102    content.push(collapse.with_duration(1.0));
103
104    r.play(
105        CameraFrame::default()
106            .show()
107            .with_duration(content.cursor_sec()),
108    );
109    r.play(content);
110}
