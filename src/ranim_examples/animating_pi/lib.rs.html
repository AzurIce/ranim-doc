16#[scene(clear_color = "#000000")]
17#[wasm_demo_doc]
18#[output(dir = "./output/animating_pi")]
19fn animating_pi(r: &mut RanimScene) {
20    let cam = CameraFrame::default();
21
22    let (w, h) = (10, 10);
23    let mut pis = (0..h)
24        .cartesian_product(0..w)
25        .map(|(_, _)| {
26            SvgItem::new(typst_svg("$pi$"))
27                .with(|x| {
28                    x.set_color(manim::WHITE);
29                })
30                .with(|x| {
31                    x.scale_to_min(&[
32                        ScaleHint::PorportionalY(cam.frame_height / h as f64),
33                        ScaleHint::PorportionalX(cam.frame_height / w as f64),
34                    ])
35                    .discard()
36                })
37        })
38        .collect::<Vec<_>>();
39    pis.arrange_in_grid(
40        usizevec3(10, 10, usize::MAX),
41        DVec3::splat(8.0 / 10.0),
42        DVec3::splat(0.2),
43    )
44    .move_to(DVec3::ZERO);
45
46    let mut vitems = pis
47        .into_iter()
48        .flat_map(Vec::<VItem>::from)
49        .collect::<Vec<_>>();
50    vitems
51        .scale_to(ScaleHint::PorportionalY(TAU - 0.25))
52        .move_to(DVec3::ZERO);
53
54    let mut content = seq![vitems.clone().show()];
55    content
56        .hold(1.0)
57        .push(
58            vitems
59                .morph(|x| {
60                    x.apply_complex_map(|c| c.exp());
61                })
62                .with_duration(5.0)
63                .with_rate_func(smooth),
64        )
65        .hold(1.0)
66        .push(
67            vitems
68                .morph(|x| {
69                    x.apply_point_func(|p| {
70                        p.x += 0.5 * p.y.sin();
71                        p.y += 0.5 * p.x.cos();
72                    });
73                })
74                .with_duration(5.0)
75                .with_rate_func(smooth),
76        )
77        .hold(1.0);
78
79    r.play(cam.show().with_duration(content.cursor_sec()));
80    r.play(content);
81    r.insert_time_mark(5.0, TimeMark::Capture("preview.png".to_string()));
82}
