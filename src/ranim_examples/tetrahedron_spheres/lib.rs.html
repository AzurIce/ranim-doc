1use std::f64::consts::{PI, TAU};
2
3use ranim::{
4    color::palettes::manim,
5    glam::DVec3,
6    items::mesh::{Sphere, Surface},
7    prelude::*,
8    utils::rate_functions::linear,
9};
10// Custom animation: rotate each sphere's rigid placement around the group
11// center on the Z axis. The orbit is composed outward from the vertex offset:
12// `T(group_center) ∘ Rz(angle) ∘ T(vertex_offset)`.
13struct RotateAroundZ {
14    src: Transformed<Surface, Rigid>,
15    group_center: DVec3,
16    vertex_offset: DVec3,
17    total_angle: f64,
18}
19
20impl Eval for RotateAroundZ {
21    type Output = Transformed<Surface, Rigid>;
22
23    fn eval_alpha(&self, alpha: f64) -> Self::Output {
24        let angle = self.total_angle * alpha;
25        let mut result = self.src.clone();
26        result.transform = Rigid::from_translation(self.group_center)
27            .compose(&Rigid::from_axis_angle(DVec3::Z, angle))
28            .compose(&Rigid::from_translation(self.vertex_offset));
29        result
30    }
31}

33#[scene]
34#[wasm_demo_doc]
35#[output(dir = "./output/tetrahedron_spheres")]
36fn tetrahedron_spheres(r: &mut RanimScene) {
37    let phi = 60.0 * PI / 180.0;
38    let theta = PI / 2.0;
39    let distance = 6.0;
40
41    // Fixed camera, Z-up
42    let cam = CameraFrame::from_spherical(phi, theta, distance);
43
44    // Regular tetrahedron centered at the origin, edge length `a`.
45    // Z is the vertical axis; the three base vertices share the same Z coordinate.
46    let a: f64 = 2.5;
47    let sqrt6 = 6.0_f64.sqrt();
48    let sqrt3 = 3.0_f64.sqrt();
49
50    let vertices = [
51        DVec3::new(0.0, 0.0, a * sqrt6 / 4.0),               // apex
52        DVec3::new(0.0, a * sqrt3 / 3.0, -a * sqrt6 / 12.0), // base
53        DVec3::new(-a / 2.0, -a * sqrt3 / 6.0, -a * sqrt6 / 12.0), // base
54        DVec3::new(a / 2.0, -a * sqrt3 / 6.0, -a * sqrt6 / 12.0), // base
55    ];
56
57    let colors = [manim::BLUE_C, manim::RED_C, manim::GREEN_C, manim::YELLOW_C];
58
59    let radius = 0.6 * a;
60    let resolution = (31, 16);
61    let x_offset = 3.5;
62
63    let mut left_group = AnimStack::new();
64
65    // Left group: smooth shading (analytical normals from Sphere)
66    let left_center = DVec3::new(-x_offset, 0.0, 0.0);
67    for (vertex, color) in vertices.iter().zip(colors.iter()) {
68        let mut surface = Surface::from(
69            Sphere::new(radius)
70                .with_resolution(resolution)
71                .with_fill_color(color.with_alpha(0.5)),
72        )
73        .with_smooth_normals()
74        .transformed(Rigid::from_translation(left_center + *vertex));
75
76        left_group.push(
77            RotateAroundZ {
78                src: surface.clone(),
79                group_center: left_center,
80                vertex_offset: *vertex,
81                total_angle: TAU,
82            }
83            .apply_to(&mut surface)
84            .with_duration(8.0)
85            .with_rate_func(linear),
86        );
87    }
88
89    // Right group: flat shading (normals zeroed out)
90    let right_center = DVec3::new(x_offset, 0.0, 0.0);
91    let mut right_group = AnimStack::new();
92    for (vertex, color) in vertices.iter().zip(colors.iter()) {
93        let mut surface = Surface::from(
94            Sphere::new(radius)
95                .with_resolution(resolution)
96                .with_fill_color(color.with_alpha(0.5)),
97        )
98        .transformed(Rigid::from_translation(right_center + *vertex));
99
100        right_group.push(
101            RotateAroundZ {
102                src: surface.clone(),
103                group_center: right_center,
104                vertex_offset: *vertex,
105                total_angle: TAU,
106            }
107            .apply_to(&mut surface)
108            .with_duration(8.0)
109            .with_rate_func(linear),
110        );
111    }
112
113    let total_secs = left_group.duration_secs().max(right_group.duration_secs());
114    r.play(cam.show().with_duration(total_secs));
115    r.play(stack![left_group, right_group]);
116
117    r.insert_time_mark(total_secs, TimeMark::Capture("preview.png".to_string()));
118}
