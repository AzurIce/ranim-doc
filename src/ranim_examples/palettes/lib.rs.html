1use ranim::{
2    color::palettes::manim::*,
3    glam::{dvec2, dvec3},
4    items::vitem::{VItem, geometry::Rectangle},
5    prelude::*,
6};
7
8#[scene]
9#[wasm_demo_doc]
10#[output(dir = "./output/palettes")]
11fn palettes(r: &mut RanimScene) {
12    let frame_size = dvec2(8.0 * 16.0 / 9.0, 8.0);
13    let padded_frame_size = frame_size * 0.9;
14
15    let colors = vec![
16        vec![BLUE_E, BLUE_D, BLUE_C, BLUE_B, BLUE_A],
17        vec![TEAL_E, TEAL_D, TEAL_C, TEAL_B, TEAL_A],
18        vec![GREEN_E, GREEN_D, GREEN_C, GREEN_B, GREEN_A],
19        vec![YELLOW_E, YELLOW_D, YELLOW_C, YELLOW_B, YELLOW_A],
20        vec![GOLD_E, GOLD_D, GOLD_C, GOLD_B, GOLD_A],
21        vec![RED_E, RED_D, RED_C, RED_B, RED_A],
22        vec![MAROON_E, MAROON_D, MAROON_C, MAROON_B, MAROON_A],
23        vec![PURPLE_E, PURPLE_D, PURPLE_C, PURPLE_B, PURPLE_A],
24        vec![GREY_E, GREY_D, GREY_C, GREY_B, GREY_A],
25        vec![WHITE, BLACK, GREEN_SCREEN],
26        vec![GREY_BROWN, LIGHT_BROWN, PINK, LIGHT_PINK, ORANGE],
27    ];
28
29    let padded_frame_start = dvec2(padded_frame_size.x / -2.0, padded_frame_size.y / -2.0);
30    let h_step = padded_frame_size.y / colors.len() as f64;
31
32    let squares = colors
33        .iter()
34        .enumerate()
35        .flat_map(|(i, row)| {
36            let y = i as f64 * h_step;
37            let w_step = padded_frame_size.x / row.len() as f64;
38            row.iter().enumerate().map(move |(j, color)| {
39                let x = j as f64 * w_step;
40                let mut rect: VItem = Rectangle::new(w_step, h_step)
41                    .with(|rect| {
42                        rect.stroke_width = 0.0;
43                        rect.set_color(*color);
44                    })
45                    .into();
46                rect.apply(Translation(
47                    padded_frame_start.extend(0.0) + dvec3(x + w_step * 0.5, y + h_step * 0.5, 0.0),
48                ));
49                rect
50            })
51        })
52        .collect::<Vec<_>>();
53    let total_secs = 0.01;
54    r.play(CameraFrame::default().show().with_duration(total_secs));
55    r.play(squares.show().with_duration(total_secs));
56    r.insert_time_mark(0.0, TimeMark::Capture("preview.png".to_string()));
57}
