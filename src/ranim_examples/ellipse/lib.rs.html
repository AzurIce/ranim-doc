1use ranim::{
2    anims::{creation::WritingAnim, fading::FadingAnim, morph::MorphAnim},
3    color::palettes::manim,
4    core::animation::StaticAnim,
5    glam::dvec2,
6    items::vitem::{
7        VItem,
8        geometry::{Circle, Ellipse},
9    },
10    prelude::*,
11    utils::rate_functions::smooth,
12};
13
14#[scene]
15#[wasm_demo_doc]
16#[output(dir = "./output/ellipse")]
17fn ellipse(r: &mut RanimScene) {
18    // Create an ellipse (semi-axes: 3.0 x 1.5) and write it in
19    let ellipse = Ellipse::new(dvec2(3.0, 1.5)).with(|e| {
20        e.set_stroke_color(manim::BLUE_C);
21    });
22    let mut vitem = VItem::from(ellipse);
23
24    let mut content = seq![vitem.write().with_rate_func(smooth)];
25
26    // Transform to a circle
27    let circle = Circle::new(2.0).with(|c| {
28        c.set_stroke_color(manim::RED_C);
29    });
30    content.push(vitem.morph_to(VItem::from(circle)).with_rate_func(smooth));
31
32    // Transform to a vertical ellipse
33    let ellipse_v = Ellipse::new(dvec2(1.0, 2.5)).with(|e| {
34        e.set_stroke_color(manim::GREEN_C);
35    });
36    content.push(
37        vitem
38            .morph_to(VItem::from(ellipse_v))
39            .with_rate_func(smooth),
40    );
41
42    // Fade out
43    content.push(vitem.fade_out().with_rate_func(smooth));
44
45    r.play(
46        CameraFrame::default()
47            .show()
48            .with_duration(content.cursor_sec()),
49    );
50    r.play(content);
51
52    r.insert_time_mark(1.0, TimeMark::Capture("preview.png".to_string()));
53}
