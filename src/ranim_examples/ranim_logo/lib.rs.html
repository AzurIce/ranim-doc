19fn build_logo(logo_width: f64) -> [VItem; 6] {
20    // The logo layout is intrinsic art coordinates, so the placement is
21    // baked into point data right after converting to VItems.
22    let red_bg_rect = VItem::from(Rectangle::new(logo_width / 2.0, logo_width)).with(|rect| {
23        rect.set_color(manim::RED_C.with_alpha(0.5))
24            .move_to(dvec3(-logo_width / 4.0, 0.0, 0.0));
25    });
26    let red_rect = VItem::from(Rectangle::new(logo_width / 4.0, logo_width)).with(|rect| {
27        rect.set_color(manim::RED_C).move_anchor_to(
28            AabbPoint(dvec3(1.0, 0.0, 0.0)),
29            dvec3(-logo_width / 4.0, 0.0, 0.0),
30        );
31    });
32
33    let green_bg_sq = VItem::from(Square::new(logo_width / 2.0)).with(|sq| {
34        sq.set_color(manim::GREEN_C.with_alpha(0.5)).move_to(dvec3(
35            logo_width / 4.0,
36            logo_width / 4.0,
37            0.0,
38        ));
39    });
40    let green_triangle = VItem::from(Polygon::new(vec![
41        dvec3(0.0, logo_width / 2.0, 0.0),
42        dvec3(logo_width / 2.0, logo_width / 2.0, 0.0),
43        dvec3(logo_width / 2.0, 0.0, 0.0),
44    ]))
45    .with(|tri| {
46        tri.set_color(manim::GREEN_C);
47    }); // ◥
48
49    let blue_bg_sq = VItem::from(Square::new(logo_width / 2.0)).with(|sq| {
50        sq.set_color(manim::BLUE_C.with_alpha(0.5)).move_to(dvec3(
51            logo_width / 4.0,
52            -logo_width / 4.0,
53            0.0,
54        ));
55    });
56    let blue_triangle = green_triangle.clone().with(|tri| {
57        tri.set_color(manim::BLUE_C);
58        tri.with_origin(AabbPoint::CENTER, |x| {
59            x.rotate_on_z(PI);
60        });
61        tri.shift(DVec3::NEG_Y * logo_width / 2.0);
62    }); // ◣
63
64    [
65        red_bg_rect,
66        red_rect,
67        green_bg_sq,
68        green_triangle,
69        blue_bg_sq,
70        blue_triangle,
71    ]
72}

76fn ranim_logo(r: &mut RanimScene) {
77    let frame_size = dvec2(8.0 * 16.0 / 9.0, 8.0);
78    let logo_width = frame_size.y * 0.618;
79
80    let logo = build_logo(logo_width);
81    let mut logo_parts = logo.map(|mut item| {
82        (
83            seq![item.write().with_duration(3.0).with_rate_func(smooth)],
84            item,
85        )
86    });
87
88    let mut ranim_text = Vec::<VItem>::from(
89        SvgItem::new(typst_svg(
90            r#"
91#align(center)[
92    #text(10pt, font: "LXGW Bright")[Ranim]
93]"#,
94        ))
95        .with(|text| {
96            text.set_color(manim::WHITE)
97                .scale_to(ScaleHint::PorportionalY(1.0))
98                .move_to(DVec3::NEG_Y * 2.5);
99        }),
100    );
101    let gap_ratio = 1.0 / 60.0;
102    let gap = logo_width * gap_ratio;
103    let scale = (logo_width - gap * 2.0) / logo_width;
104    let scale = [
105        dvec3(scale, 1.0, 1.0),
106        dvec3(scale, scale, 1.0),
107        dvec3(scale, scale, 1.0),
108    ];
109    let anchor = [
110        AabbPoint(dvec3(-1.0, 0.0, 0.0)),
111        AabbPoint(dvec3(1.0, 1.0, 0.0)),
112        AabbPoint(dvec3(1.0, -1.0, 0.0)),
113    ];
114    logo_parts
115        .iter_mut()
116        .chunks(2)
117        .into_iter()
118        .zip(scale.into_iter().zip(anchor))
119        .for_each(|(chunk, (scale, anchor))| {
120            chunk.for_each(|(sequence, item)| {
121                sequence.push(
122                    item.morph(|data| {
123                        data.with_origin(anchor, |x| {
124                            x.scale(scale);
125                        });
126                        data.with_origin(DVec3::ZERO, |x| {
127                            x.scale(dvec3(0.9, 0.9, 1.0));
128                        });
129                        data.shift(dvec3(0.0, 1.3, 0.0));
130                    })
131                    .with_rate_func(smooth),
132                );
133            })
134        });
135    let mut text_sequence = AnimSequence::new();
136    text_sequence.forward(3.5).push(
137        ranim_text
138            .iter_mut()
139            .map(|item| item.write())
140            .into_lagged(0.2)
141            .with_duration(2.0),
142    );
143
144    let phase_end = logo_parts
145        .iter()
146        .map(|(sequence, _)| sequence.cursor_sec())
147        .chain(std::iter::once(text_sequence.cursor_sec()))
148        .fold(0.0, f64::max);
149    logo_parts
150        .iter_mut()
151        .for_each(|(sequence, _)| _ = sequence.hold_to(phase_end));
152    text_sequence.hold_to(phase_end);
153
154    r.insert_time_mark(phase_end, TimeMark::Capture("preview.png".to_string()));
155    logo_parts
156        .iter_mut()
157        .for_each(|(sequence, _)| _ = sequence.hold(1.0));
158    text_sequence.hold(1.0);
159
160    logo_parts.iter_mut().for_each(|(sequence, item)| {
161        sequence.push(item.unwrite().with_duration(3.0).with_rate_func(smooth));
162    });
163    text_sequence.push(
164        ranim_text
165            .iter_mut()
166            .map(|item| item.unwrite())
167            .into_lagged(0.0),
168    );
169
170    let total_secs = logo_parts
171        .iter()
172        .map(|(sequence, _)| sequence.cursor_sec())
173        .chain(std::iter::once(text_sequence.cursor_sec()))
174        .fold(0.0, f64::max);
175    let mut content = AnimStack::new();
176    for (sequence, _) in logo_parts {
177        content.push(sequence);
178    }
179    r.play(CameraFrame::default().show().with_duration(total_secs));
180    r.play(stack![content, text_sequence]);
181}
