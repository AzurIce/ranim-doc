12fn selective_sort(r: &mut RanimScene, num: usize) {
13    let frame_size = dvec2(8.0 * 16.0 / 9.0, 8.0);
14    let padded_frame_size = frame_size * 0.9;
15
16    let anim_step_duration = 15.0 / num.pow(2) as f64;
17
18    let width_unit = padded_frame_size.x / num as f64;
19    let height_unit = padded_frame_size.y / num as f64;
20
21    let mut rng = rand_chacha::ChaChaRng::seed_from_u64(114514);
22    let mut heights = (1..=num)
23        .map(|x| x as f64 * height_unit)
24        .collect::<Vec<f64>>();
25    heights.shuffle(&mut rng);
26
27    let padded_frame_bl = dvec2(padded_frame_size.x / -2.0, padded_frame_size.y / -2.0);
28    let mut rects = heights
29        .iter()
30        .enumerate()
31        .map(|(i, &height)| {
32            let target_bc_coord =
33                padded_frame_bl.extend(0.0) + DVec3::X * (width_unit * i as f64 + width_unit / 2.0);
34            let rect = Rectangle::new(width_unit, height).with(|rect| {
35                rect.fill_rgba = manim::WHITE.with_alpha(0.5);
36                rect.scale_axes(dvec2(0.8, 0.8));
37            });
38            let mut rect = VItem::from(rect);
39            rect.move_anchor_to(AabbPoint(dvec3(0.0, -1.0, 0.0)), target_bc_coord);
40            let sequence = seq![rect.show()];
41            (sequence, rect)
42        })
43        .collect::<Vec<_>>();
44
45    let highlight = |rect: &mut VItem| {
46        rect.morph(|data| {
47            data.set_color(manim::RED_C).set_fill_opacity(0.5);
48        })
49        .with_duration(anim_step_duration)
50        .with_rate_func(linear)
51    };
52    let unhighlight = |rect: &mut VItem| {
53        rect.morph(|data| {
54            data.set_color(manim::WHITE).set_fill_opacity(0.5);
55        })
56        .with_duration(anim_step_duration)
57        .with_rate_func(linear)
58    };
59
60    let shift_right = DVec3::X * width_unit;
61    let sync = |rects: &mut [(AnimSequence, VItem)]| {
62        let target = rects
63            .iter()
64            .map(|(sequence, _)| sequence.cursor_sec())
65            .fold(0.0, f64::max);
66        rects.iter_mut().for_each(|(sequence, _)| {
67            sequence.hold_to(target);
68        });
69    };
70    for i in 0..num - 1 {
71        let (sequence, rect) = &mut rects[i];
72        sequence.push(highlight(rect));
73        for j in i + 1..num {
74            let (sequence, rect) = &mut rects[j];
75            sequence.push(highlight(rect));
76            sync(&mut rects);
77
78            if heights[i] > heights[j] {
79                let dir = [shift_right, -shift_right];
80                let color = [manim::BLUE_C, manim::RED_C];
81                rects
82                    .get_disjoint_mut([i, j])
83                    .unwrap()
84                    .into_iter()
85                    .zip(dir)
86                    .zip(color)
87                    .for_each(|(((sequence, rect), dir), color)| {
88                        sequence.push(
89                            rect.morph(|rect| {
90                                rect.shift(dir * (j - i) as f64)
91                                    .set_color(color)
92                                    .set_fill_opacity(0.5);
93                            })
94                            .with_duration(anim_step_duration)
95                            .with_rate_func(linear),
96                        );
97                    });
98                heights.swap(i, j);
99                rects.swap(i, j);
100            }
101            let (sequence, rect) = &mut rects[j];
102            sequence.push(unhighlight(rect));
103            sync(&mut rects);
104        }
105        let (sequence, rect) = &mut rects[i];
106        sequence.push(unhighlight(rect));
107    }
108    sync(&mut rects);
109
110    let total_secs = rects[0].0.cursor_sec();
111    r.insert_time_mark(
112        total_secs / 2.0,
113        TimeMark::Capture(format!("preview-{num}.png")),
114    );
115
116    let mut content = AnimStack::new();
117    for (sequence, _) in rects {
118        content.push(sequence);
119    }
120    r.play(CameraFrame::default().show().with_duration(total_secs));
121    r.play(content);
122}
