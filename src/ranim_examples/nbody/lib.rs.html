1//! N-body gravitational simulation — a canonical "iterative only" animation:
2//! no closed-form solution, chaotic, and inherently stateful across frames.
3//!
4//! `NBodyState::new(n)` places `n` equal masses on a regular n-gon with
5//! tangential velocities (exact circular speed for the relative equilibrium,
6//! plus a tiny asymmetry). The logical simulation duration is the local
7//! `sim_secs` value captured by the step closure (not a global constant), and
8//! the same value is passed to `with_duration`. Each body leaves a fading
9//! trail of recent positions; the `Extract` impl projects bodies and trails
10//! once per frame. Circles stay semantic and their positions live in
11//! `Translation` wrappers applied at extraction time.
12//!
13//! Behavior by `n`: n = 3 stays inside the frame and wobbles chaotically for
14//! the whole scene; n >= 4 destabilizes and ejects a body (a dramatic finale).
15
16use ranim::{
17    color::{AlphaColor, Srgb, palettes::manim},
18    core::Extract,
19    core::animation::eval::iterative::Iterative,
20    core::core_item::CoreItem,
21    glam::DVec3,
22    items::vitem::geometry::Circle,
23    prelude::*,
24};
25use std::f64::consts::PI;
26
27const G: f64 = 8.0;
28const TRAIL_SAMPLE_EVERY: usize = 4; // sample a trail point every 4th step (~30 Hz)
29const TRAIL_LEN: usize = 90; // ~3 s of trail per body
30
31const PALETTE: [AlphaColor<Srgb>; 6] = [
32    manim::BLUE_C,
33    manim::RED_C,
34    manim::YELLOW_C,
35    manim::GREEN_C,
36    manim::PURPLE_C,
37    manim::ORANGE,
38];
39
40#[derive(Clone, Copy)]
41struct Body {
42    pos: DVec3,
43    vel: DVec3,
44    mass: f64,
45}
46
47/// The N-body system's full state.
48#[derive(Clone)]
49struct NBodyState {
50    bodies: Vec<Body>,
51    trails: Vec<Vec<DVec3>>,
52    step_count: usize,
53    colors: Vec<AlphaColor<Srgb>>,
54}
55
56impl NBodyState {
57    /// Place `n` equal masses on a regular n-gon.
58    ///
59    /// The circular speed is exact for the regular n-gon relative equilibrium:
60    /// `v² = G·Σ csc(πj/n) / (4R)`. A 4% overspeed plus a tiny asymmetry keeps
61    /// the system lively (and chaotic — the n-gon equilibrium is unstable).
62    fn new(n: usize) -> Self {
63        assert!(n >= 2, "nbody needs at least 2 bodies");
64        let radius = 2.0;
65        let csc_sum: f64 = (1..n).map(|j| 1.0 / (PI * j as f64 / n as f64).sin()).sum();
66        let v_circ = (G * csc_sum / (4.0 * radius)).sqrt();
67        let v0 = 1.04 * v_circ;
68
69        let mut bodies = Vec::with_capacity(n);
70        for i in 0..n {
71            let angle = PI * 2.0 * i as f64 / n as f64;
72            let pos = DVec3::new(radius * angle.cos(), radius * angle.sin(), 0.0);
73            let velocity = v0 * (1.0 + 0.001 * i as f64);
74            let vel = DVec3::new(-velocity * angle.sin(), velocity * angle.cos(), 0.0);
75            bodies.push(Body {
76                pos,
77                vel,
78                mass: 1.0,
79            });
80        }
81
82        let colors = (0..n).map(|i| PALETTE[i % PALETTE.len()]).collect();
83        Self {
84            bodies,
85            trails: vec![Vec::new(); n],
86            step_count: 0,
87            colors,
88        }
89    }
90
91    fn accelerations(&self) -> Vec<DVec3> {
92        let mut accs = vec![DVec3::ZERO; self.bodies.len()];
93        for (i, acc) in accs.iter_mut().enumerate() {
94            for j in 0..self.bodies.len() {
95                if i == j {
96                    continue;
97                }
98                let r = self.bodies[j].pos - self.bodies[i].pos;
99                let d2 = r.length_squared();
100                let d = d2.sqrt();
101                *acc += r * (G * self.bodies[j].mass / (d2 * d));
102            }
103        }
104        accs
105    }
106
107    fn step(&mut self, dt: f64) {
108        // Velocity Verlet (symplectic): conserves energy far better than
109        // semi-implicit Euler for gravitational orbits.
110        let a0 = self.accelerations();
111        for (i, body) in self.bodies.iter_mut().enumerate() {
112            body.pos += body.vel * dt + a0[i] * (0.5 * dt * dt);
113        }
114        let a1 = self.accelerations();
115        for (i, body) in self.bodies.iter_mut().enumerate() {
116            body.vel += (a0[i] + a1[i]) * (0.5 * dt);
117        }
118
119        self.step_count += 1;
120        if self.step_count.is_multiple_of(TRAIL_SAMPLE_EVERY) {
121            for (i, trail) in self.trails.iter_mut().enumerate() {
122                trail.push(self.bodies[i].pos);
123                if trail.len() > TRAIL_LEN {
124                    trail.remove(0);
125                }
126            }
127        }
128    }
129}
130
131impl Extract for NBodyState {
132    type Target = CoreItem;
133
134    fn extract_into(&self, buf: &mut Vec<CoreItem>) {
135        // Trails: dim dots at recent positions.
136        for (i, trail) in self.trails.iter().enumerate() {
137            for &p in trail {
138                let mut dot = Circle::new(0.02).transformed(Translation(p));
139                dot.set_fill_color(self.colors[i]);
140                dot.set_fill_opacity(0.15);
141                dot.set_stroke_opacity(0.0);
142                dot.extract_into(buf);
143            }
144        }
145        // Bodies.
146        for (i, body) in self.bodies.iter().enumerate() {
147            let mut ball = Circle::new(0.32).transformed(Translation(body.pos));
148            ball.set_fill_color(self.colors[i]);
149            ball.set_stroke_opacity(0.0);
150            ball.extract_into(buf);
151        }
152    }
153}

155#[scene]
156#[wasm_demo_doc]
157#[output(dir = "./output/nbody")]
158fn nbody(r: &mut RanimScene) {
159    let sim_secs = 32.0;
160    r.play(CameraFrame::default().show().with_duration(sim_secs));
161    // n = 3: chaotic wobble that stays in frame; n >= 4: ejection finale.
162    r.play(
163        Iterative::from_fn(
164            NBodyState::new(3),
165            move |state: &mut NBodyState, _alpha: f64, delta_alpha: f64| {
166                state.step(sim_secs * delta_alpha);
167            },
168        )
169        .with_steps((sim_secs * 60.0) as usize)
170        .with_duration(sim_secs),
171    );
172    r.insert_time_mark(sim_secs / 2.0, TimeMark::Capture("preview.png".to_string()));
173}
