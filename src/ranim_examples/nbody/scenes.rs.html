1//! N-body gravitational simulation — a canonical "iterative only" animation:
2//! no closed-form solution, chaotic, and inherently stateful across frames.
3//!
4//! `NBodyState::new(n)` places `n` equal masses on a regular n-gon with
5//! tangential velocities (exact circular speed for the relative equilibrium,
6//! plus a tiny asymmetry). The logical simulation duration is the local
7//! `sim_secs` value captured by the step closure (not a global constant), and
8//! the same value is passed to `with_duration`. Each body leaves a fading
9//! trail of recent positions; the `Extract` impl projects bodies and trails
10//! once per frame. Circles stay semantic and their positions live in
11//! `Translation` wrappers applied at extraction time.
12//!
13//! Behavior by `n`: n = 3 stays inside the frame and wobbles chaotically for
14//! the whole scene; n >= 4 destabilizes and ejects a body (a dramatic finale).
15
16use ranim::{
17    color::{AlphaColor, Srgb, palettes::manim},
18    core::animation::eval::iterative::Iterative,
19    glam::DVec3,
20    items::vitem::geometry::Circle,
21    prelude::*,
22};
23use std::f64::consts::PI;
24
25const G: f64 = 8.0;
26const TRAIL_SAMPLE_EVERY: usize = 4; // sample a trail point every 4th step (~30 Hz)
27const TRAIL_LEN: usize = 90; // ~3 s of trail per body
28
29const PALETTE: [AlphaColor<Srgb>; 6] = [
30    manim::BLUE_C,
31    manim::RED_C,
32    manim::YELLOW_C,
33    manim::GREEN_C,
34    manim::PURPLE_C,
35    manim::ORANGE,
36];
37
38#[derive(Clone, Copy)]
39struct Body {
40    pos: DVec3,
41    vel: DVec3,
42    mass: f64,
43}
44
45/// The N-body system's full state.
46#[derive(Clone)]
47struct NBodyState {
48    bodies: Vec<Body>,
49    trails: Vec<Vec<DVec3>>,
50    step_count: usize,
51    colors: Vec<AlphaColor<Srgb>>,
52}
53
54impl NBodyState {
55    /// Place `n` equal masses on a regular n-gon.
56    ///
57    /// The circular speed is exact for the regular n-gon relative equilibrium:
58    /// `v² = G·Σ csc(πj/n) / (4R)`. A 4% overspeed plus a tiny asymmetry keeps
59    /// the system lively (and chaotic — the n-gon equilibrium is unstable).
60    fn new(n: usize) -> Self {
61        assert!(n >= 2, "nbody needs at least 2 bodies");
62        let radius = 2.0;
63        let csc_sum: f64 = (1..n).map(|j| 1.0 / (PI * j as f64 / n as f64).sin()).sum();
64        let v_circ = (G * csc_sum / (4.0 * radius)).sqrt();
65        let v0 = 1.04 * v_circ;
66
67        let mut bodies = Vec::with_capacity(n);
68        for i in 0..n {
69            let angle = PI * 2.0 * i as f64 / n as f64;
70            let pos = DVec3::new(radius * angle.cos(), radius * angle.sin(), 0.0);
71            let velocity = v0 * (1.0 + 0.001 * i as f64);
72            let vel = DVec3::new(-velocity * angle.sin(), velocity * angle.cos(), 0.0);
73            bodies.push(Body {
74                pos,
75                vel,
76                mass: 1.0,
77            });
78        }
79
80        let colors = (0..n).map(|i| PALETTE[i % PALETTE.len()]).collect();
81        Self {
82            bodies,
83            trails: vec![Vec::new(); n],
84            step_count: 0,
85            colors,
86        }
87    }
88
89    fn accelerations(&self) -> Vec<DVec3> {
90        let mut accs = vec![DVec3::ZERO; self.bodies.len()];
91        for (i, acc) in accs.iter_mut().enumerate() {
92            for j in 0..self.bodies.len() {
93                if i == j {
94                    continue;
95                }
96                let r = self.bodies[j].pos - self.bodies[i].pos;
97                let d2 = r.length_squared();
98                let d = d2.sqrt();
99                *acc += r * (G * self.bodies[j].mass / (d2 * d));
100            }
101        }
102        accs
103    }
104
105    fn step(&mut self, dt: f64) {
106        // Velocity Verlet (symplectic): conserves energy far better than
107        // semi-implicit Euler for gravitational orbits.
108        let a0 = self.accelerations();
109        for (i, body) in self.bodies.iter_mut().enumerate() {
110            body.pos += body.vel * dt + a0[i] * (0.5 * dt * dt);
111        }
112        let a1 = self.accelerations();
113        for (i, body) in self.bodies.iter_mut().enumerate() {
114            body.vel += (a0[i] + a1[i]) * (0.5 * dt);
115        }
116
117        self.step_count += 1;
118        if self.step_count.is_multiple_of(TRAIL_SAMPLE_EVERY) {
119            for (i, trail) in self.trails.iter_mut().enumerate() {
120                trail.push(self.bodies[i].pos);
121                if trail.len() > TRAIL_LEN {
122                    trail.remove(0);
123                }
124            }
125        }
126    }
127}

129#[cfg(feature = "render")]
130impl ranim::render::extract::Extract for NBodyState {
131    fn extract(&self, ctx: &mut ranim::render::extract::ExtractContext<'_>) {
132        // Trails: dim dots at recent positions.
133        for (i, trail) in self.trails.iter().enumerate() {
134            for &p in trail {
135                let mut dot = Circle::new(0.02).transformed(Translation(p));
136                dot.set_fill_color(self.colors[i]);
137                dot.set_fill_opacity(0.15);
138                dot.set_stroke_opacity(0.0);
139                dot.extract(ctx);
140            }
141        }
142        // Bodies.
143        for (i, body) in self.bodies.iter().enumerate() {
144            let mut ball = Circle::new(0.32).transformed(Translation(body.pos));
145            ball.set_fill_color(self.colors[i]);
146            ball.set_stroke_opacity(0.0);
147            ball.extract(ctx);
148        }
149    }
150}

152#[scene]
153#[wasm_demo_doc]
154#[output(dir = "./output/nbody")]
155fn nbody(r: &mut RanimScene) {
156    let sim_secs = 32.0;
157    r.play(CameraFrame::default().show().with_duration(sim_secs));
158    // n = 3: chaotic wobble that stays in frame; n >= 4: ejection finale.
159    r.play(
160        Iterative::from_fn(
161            NBodyState::new(3),
162            move |state: &mut NBodyState, _alpha: f64, delta_alpha: f64| {
163                state.step(sim_secs * delta_alpha);
164            },
165        )
166        .with_steps((sim_secs * 60.0) as usize)
167        .with_duration(sim_secs),
168    );
169    r.insert_time_mark(sim_secs / 2.0, TimeMark::Capture("preview.png".to_string()));
170}

186        #[cfg(feature = "render")]

189
190            app.insert_resource(RenderSetup::new(|render_app| {
191                render_app.add_plugins(RanimItemsPlugin);
192                configure_render(render_app);
193            }));
194        }
195    }
