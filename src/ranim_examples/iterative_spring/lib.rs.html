1//! Iterative animation example: a damped spring (state stepped by a closure,
2//! no ECS required).
3//!
4//! The spring's state (`x`, `v`) is owned by `Iterative` and advanced by a
5//! closure via semi-implicit Euler; the `Extract` impl projects the state into
6//! a rectangle once per frame, with the displacement carried by a
7//! `Translation` wrapper. Reset is structural — the adapter restores the
8//! stored initial state, so `SceneEvaluator::sample_at` matches forward
9//! advancement for free.
10//!
11//! The spring's logical duration is the local `sim_secs` value captured by the
12//! closure; the same value is passed to `with_duration`, so physical time and
13//! timeline time stay in sync.
14
15use ranim::{
16    color::palettes::manim, core::Extract, core::animation::eval::iterative::Iterative,
17    core::core_item::CoreItem, glam::dvec3, items::vitem::geometry::Rectangle, prelude::*,
18};
19
20/// The spring's state: displacement and velocity.
21#[derive(Clone)]
22struct SpringState {
23    x: f64,
24    v: f64,
25}
26
27impl Extract for SpringState {
28    type Target = CoreItem;
29
30    fn extract_into(&self, buf: &mut Vec<CoreItem>) {
31        let mut ball = Rectangle::new(0.6, 0.6).transformed(Translation(dvec3(self.x, 0.0, 0.0)));
32        ball.set_fill_color(manim::BLUE_C);
33        ball.set_stroke_opacity(0.0);
34        ball.extract_into(buf);
35    }
36}
37
38const K: f64 = 25.0;
39const C: f64 = 1.0;
40
41#[scene]
42#[wasm_demo_doc]
43#[output(dir = "./output/iterative_spring")]
44fn iterative_spring(r: &mut RanimScene) {
45    let sim_secs = 4.0;
46    r.play(CameraFrame::default().show().with_duration(sim_secs));
47    r.play(
48        Iterative::from_fn(
49            SpringState { x: 1.0, v: 0.0 },
50            move |state: &mut SpringState, _alpha: f64, delta_alpha: f64| {
51                let dt = sim_secs * delta_alpha;
52                let acc = -K * state.x - C * state.v;
53                state.v += acc * dt;
54                state.x += state.v * dt;
55            },
56        )
57        .with_duration(sim_secs),
58    );
59    r.insert_time_mark(sim_secs / 2.0, TimeMark::Capture("preview.png".to_string()));
60}
