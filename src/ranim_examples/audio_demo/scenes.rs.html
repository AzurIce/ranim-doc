14/// A little arpeggio, synthesized from pure sines (no audio assets).
15fn arpeggio(notes: &[f64], note_secs: f64, repeats: usize) -> AudioClip {
16    let sample_rate = 48_000_u32;
17    let note_len = (note_secs * sample_rate as f64) as usize;
18    let mut pcm = Vec::with_capacity(notes.len() * note_len * repeats);
19    for _ in 0..repeats {
20        for &freq in notes {
21            for j in 0..note_len {
22                let t = j as f64 / sample_rate as f64;
23                // Fade each note in and out so consecutive notes don't click.
24                let env = (t / note_secs).min(1.0) * (1.0 - t / note_secs);
25                let wave = (2.0 * std::f64::consts::PI * freq * t).sin();
26                pcm.push((0.35 * env * wave) as f32);
27            }
28        }
29    }
30    AudioClip::from_pcm(pcm, sample_rate, 1)
31}

33#[scene]
34#[wasm_demo_doc]
35#[output(dir = "./output/audio_demo")]
36fn audio_demo(r: &mut RanimScene) {
37    let mut square = Square::new(2.0);
38    square.set_color(manim::BLUE_C);
39    let mut vitem = VItem::from(square.clone());
40
41    // The soundtrack lives in the same tree as the visuals: a looping
42    // arpeggio as BGM under the whole choreography, plus a "ding" placed
43    // exactly where the square finishes fading in.
44    let bgm = Sound::new(arpeggio(&[261.63, 329.63, 392.0, 523.25], 0.3, 4))
45        .with_fade_in(0.5)
46        .with_fade_out(1.0);
47    let ding = Sound::new(arpeggio(&[1046.5], 0.4, 1))
48        .with_gain(0.8)
49        .at(1.0);
50
51    let visual = stack![
52        seq![
53            square.fade_in().with_rate_func(smooth),
54            vitem.show().with_duration(1.0),
55            vitem.clone().unwrite().with_rate_func(smooth),
56            vitem.write().with_rate_func(smooth),
57            vitem.fade_out().with_rate_func(smooth),
58        ],
59        ding,
60    ];
61
62    r.play(
63        CameraFrame::default()
64            .show()
65            .with_duration(visual.duration_secs()),
66    );
67    r.play(visual);
68    r.play(bgm);
69
70    r.insert_time_mark(1.0, TimeMark::Capture("preview.png".to_string()));
71}
