23/// Synthesize the whole song into one clip: each note becomes a harmonic
24/// tone with a short attack/release envelope; notes are summed and the
25/// result is peak-normalized.
26///
27/// Thousands of dense note events belong in one pre-summed clip — a
28/// [`Sound`] leaf per note would scan the scene window once per leaf during
29/// mixing. Leaves are for sparse placement, not for event tracks.
30fn render_song_audio(song: &midi::MidiSong) -> AudioClip {
31    const SAMPLE_RATE: u32 = 48_000;
32    const ATTACK_SECS: f64 = 0.004;
33    const RELEASE_SECS: f64 = 0.04;
34    // The parser closes stuck notes at song end; cap the drone.
35    const MAX_NOTE_SECS: f64 = 5.0;
36
37    // Floor, not ceil: the clip must not run past the scene's visual window,
38    // or the sub-frame tail renders without a camera (D0002).
39    let total = (song.duration_secs * f64::from(SAMPLE_RATE)) as usize;
40    let mut pcm = vec![0.0_f64; total];
41    for note in &song.notes {
42        let freq = 440.0 * ((f64::from(note.key) - 69.0) / 12.0).exp2();
43        let amp = f64::from(note.velocity) / 127.0;
44        let note_len = (note.end_sec - note.start_sec).min(MAX_NOTE_SECS);
45        let len = (note_len * f64::from(SAMPLE_RATE)).ceil() as usize;
46        let start = (note.start_sec * f64::from(SAMPLE_RATE)).round() as usize;
47        let end = (start + len).min(total);
48        let attack = ((ATTACK_SECS * f64::from(SAMPLE_RATE)) as usize).max(1);
49        let release = ((RELEASE_SECS * f64::from(SAMPLE_RATE)) as usize).max(1);
50        for (j, sample) in pcm[start..end].iter_mut().enumerate() {
51            let mut envelope = amp;
52            if j < attack {
53                envelope *= j as f64 / attack as f64;
54            }
55            if j > len.saturating_sub(release) {
56                envelope *= (len - j) as f64 / release as f64;
57            }
58            let t = j as f64 / f64::from(SAMPLE_RATE);
59            let wave = 0.6 * (TAU * freq * t).sin()
60                + 0.25 * (TAU * 2.0 * freq * t).sin()
61                + 0.15 * (TAU * 3.0 * freq * t).sin();
62            *sample += envelope * wave;
63        }
64    }
65
66    let peak = pcm
67        .iter()
68        .fold(0.0_f64, |max, sample| max.max(sample.abs()));
69    let scale = if peak > 0.8 { 0.8 / peak } else { 1.0 };
70    AudioClip::from_pcm(
71        pcm.into_iter()
72            .map(|sample| (sample * scale) as f32)
73            .collect::<Vec<_>>(),
74        SAMPLE_RATE,
75        1,
76    )
77}

79fn common_layers(song: &Arc<midi::MidiSong>, layout: PianoLayout) -> AnimStack {
80    let duration = song.duration_secs;
81    let background = make_background(layout);
82
83    stack![
84        background.show().with_duration(duration),
85        PianoKeyboardEval {
86            song: song.clone(),
87            layout,
88        }
89        .with_duration(duration)
90        .with_rate_func(linear),
91        HitEffectsEval {
92            song: song.clone(),
93            layout,
94        }
95        .with_duration(duration)
96        .with_rate_func(linear),
97    ]
98}

100fn camera(duration: f64) -> impl IntoAnimNode {
101    CameraFrame {
102        pos: DVec3::ZERO,
103        frame_height: FRAME_HEIGHT,
104        ..Default::default()
105    }
106    .show()
107    .with_duration(duration)
108}

110#[scene(clear_color = "#070711")]
111#[wasm_demo_doc]
112#[output(fps = 30, dir = "./output/midi_visualizer")]
113fn midi_visualizer(r: &mut RanimScene) {
114    let song = Arc::new(parse_song());
115    let layout = PianoLayout::new(&song);
116    let duration = song.duration_secs;
117
118    r.play(camera(duration));
119    r.play(Sound::new(render_song_audio(&song)));
120    r.play(stack![
121        MidiNotesEval {
122            song: song.clone(),
123            layout,
124        }
125        .with_duration(duration)
126        .with_rate_func(linear),
127        common_layers(&song, layout),
128    ]);
129    r.insert_time_mark(duration * 0.5, TimeMark::Capture("preview.png".to_string()));
130}

132#[scene(clear_color = "#070711")]
133#[wasm_demo_doc]
134#[output(fps = 30, dir = "./output/midi_visualizer")]
135fn midi_visualizer_per_note(r: &mut RanimScene) {
136    let song = Arc::new(parse_song());
137    let layout = PianoLayout::new(&song);
138
139    let mut notes = AnimStack::new();
140    for &note in &song.notes {
141        let note_eval = SingleNoteEval::new(note, layout);
142        let start_sec = note_eval.start_sec();
143        let duration_secs = note_eval.duration_secs();
144        notes.push(
145            note_eval
146                .with_duration(duration_secs)
147                .with_rate_func(linear)
148                .at(start_sec),
149        );
150    }
151    r.play(camera(song.duration_secs));
152    r.play(Sound::new(render_song_audio(&song)));
153    r.play(stack![notes, common_layers(&song, layout)]);
154    r.insert_time_mark(
155        song.duration_secs * 0.5,
156        TimeMark::Capture("preview.png".to_string()),
157    );
158}

167#[test]
168fn every_rendered_frame_has_a_camera() {
169    // Regression: the synthesized clip is ceil-rounded up to whole samples,
170    // which used to extend the scene duration a fraction of a frame past the
171    // camera's window, leaving the last rendered frame camera-less (D0002).
172    let song = Arc::new(parse_song());
173    let layout = PianoLayout::new(&song);
174    let mut r = RanimScene::new();
175    r.play(camera(song.duration_secs));
176    r.play(Sound::new(render_song_audio(&song)));
177    r.play(stack![
178        MidiNotesEval {
179            song: song.clone(),
180            layout
181        }
182        .with_duration(song.duration_secs)
183        .with_rate_func(linear),
184        common_layers(&song, layout),
185    ]);
186    let sealed = r.seal();
187    let fps = 30.0;
188    let raw = sealed.total_secs() * fps;
189    let n = raw.ceil() as u64;
190    let num_frames = if (raw - raw.round()).abs() < 1e-9 {
191        n
192    } else {
193        n + 1
194    };
195    for f in 0..num_frames {
196        let sec = (f as f64 / fps).min(sealed.total_secs());
197        let cameras = sealed
198            .eval_at_sec(sec)
199            .filter(|(_, item)| matches!(item, ranim::core::core_item::CoreItem::CameraFrame(_)))
200            .count();
201        assert_eq!(cameras, 1, "frame {f} at {sec}s has {cameras} cameras");
202    }
