32fn close_note(
33    active: &mut HashMap<(u8, u8), Vec<(u64, u8)>>,
34    channel: u8,
35    key: u8,
36    end_tick: u64,
37    track: usize,
38    notes: &mut Vec<RawNote>,
39) {
40    let Some(starts) = active.get_mut(&(channel, key)) else {
41        return;
42    };
43    let Some((start_tick, velocity)) = starts.pop() else {
44        return;
45    };
46    notes.push(RawNote {
47        start_tick,
48        end_tick: end_tick.max(start_tick + 1),
49        key,
50        velocity,
51        track,
52    });
53}

55fn tick_to_seconds(tick: u64, tempos: &[(u64, u32)], ticks_per_beat: u64) -> f64 {
56    let mut seconds = 0.0;
57    let mut cursor_tick = 0;
58    let mut micros_per_beat = 500_000_u32;
59
60    for &(tempo_tick, next_tempo) in tempos {
61        if tempo_tick > tick {
62            break;
63        }
64        seconds += (tempo_tick - cursor_tick) as f64 * micros_per_beat as f64
65            / ticks_per_beat as f64
66            / 1_000_000.0;
67        cursor_tick = tempo_tick;
68        micros_per_beat = next_tempo;
69    }
70
71    seconds
72        + (tick - cursor_tick) as f64 * micros_per_beat as f64 / ticks_per_beat as f64 / 1_000_000.0
73}

75pub(crate) fn parse_song() -> MidiSong {
76    let smf = Smf::parse(MIDI_BYTES).expect("the bundled MIDI file should be valid");
77    let ticks_per_beat = match smf.header.timing {
78        Timing::Metrical(ticks) => u64::from(ticks.as_int()),
79        Timing::Timecode(_, _) => panic!("timecode-based MIDI is not supported by this example"),
80    };
81
82    let mut tempo_map = BTreeMap::new();
83    let mut raw_notes = Vec::new();
84    let mut song_end_tick = 0;
85
86    for (track_index, track) in smf.tracks.iter().enumerate() {
87        let mut tick = 0_u64;
88        let mut active = HashMap::<(u8, u8), Vec<(u64, u8)>>::new();
89
90        for event in track {
91            tick += u64::from(event.delta.as_int());
92            song_end_tick = song_end_tick.max(tick);
93
94            match event.kind {
95                TrackEventKind::Meta(MetaMessage::Tempo(tempo)) => {
96                    tempo_map.insert(tick, tempo.as_int());
97                }
98                TrackEventKind::Midi { channel, message } => match message {
99                    MidiMessage::NoteOn { key, vel } if vel.as_int() > 0 => {
100                        active
101                            .entry((channel.as_int(), key.as_int()))
102                            .or_default()
103                            .push((tick, vel.as_int()));
104                    }
105                    MidiMessage::NoteOn { key, .. } | MidiMessage::NoteOff { key, .. } => {
106                        close_note(
107                            &mut active,
108                            channel.as_int(),
109                            key.as_int(),
110                            tick,
111                            track_index,
112                            &mut raw_notes,
113                        );
114                    }
115                    _ => {}
116                },
117                _ => {}
118            }
119        }
120
121        for ((_, key), starts) in active {
122            for (start_tick, velocity) in starts {
123                raw_notes.push(RawNote {
124                    start_tick,
125                    end_tick: song_end_tick.max(start_tick + 1),
126                    key,
127                    velocity,
128                    track: track_index,
129                });
130            }
131        }
132    }
133
134    let tempos = tempo_map.into_iter().collect::<Vec<_>>();
135    let duration_secs = tick_to_seconds(song_end_tick, &tempos, ticks_per_beat);
136    let mut notes = raw_notes
137        .into_iter()
138        .map(|note| Note {
139            start_sec: tick_to_seconds(note.start_tick, &tempos, ticks_per_beat),
140            end_sec: tick_to_seconds(note.end_tick, &tempos, ticks_per_beat),
141            key: note.key,
142            velocity: note.velocity,
143            track: note.track,
144        })
145        .collect::<Vec<_>>();
146    notes.sort_by(|a, b| a.start_sec.total_cmp(&b.start_sec));
147
148    let mut notes_by_key = vec![Vec::new(); 128];
149    for &note in &notes {
150        notes_by_key[usize::from(note.key)].push(note);
151    }
152    let max_note_duration_secs = notes
153        .iter()
154        .map(|note| note.end_sec - note.start_sec)
155        .fold(0.0_f64, f64::max);
156
157    MidiSong {
158        notes,
159        notes_by_key,
160        max_note_duration_secs,
161        duration_secs,
162    }
163}
