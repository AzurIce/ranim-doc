11const NOTE_SPEED: f64 = 5.5;
12pub(crate) const FRAME_HEIGHT: f64 = 8.0;
13const FRAME_WIDTH: f64 = FRAME_HEIGHT * 16.0 / 9.0;
14const FRAME_LEFT: f64 = -FRAME_WIDTH * 0.5;
15const FRAME_RIGHT: f64 = FRAME_WIDTH * 0.5;
16const KEYBOARD_WIDTH: f64 = 1.45;
17const HIT_X: f64 = FRAME_LEFT + KEYBOARD_WIDTH;
18const KEY_ATTACK_SECS: f64 = 0.055;
19const KEY_RELEASE_SECS: f64 = 0.12;
20const HIT_EFFECT_SECS: f64 = 0.22;

22fn rectangle(
23    width: f64,
24    height: f64,
25    position: DVec3,
26    color: AlphaColor<Srgb>,
27    opacity: f32,
28) -> VItem {
29    let mut rectangle: VItem = Rectangle::new(width, height)
30        .with(|rectangle| {
31            rectangle
32                .set_stroke_opacity(0.0)
33                .set_fill_color(color)
34                .set_fill_opacity(opacity);
35        })
36        .into();
37    rectangle.shift(position);
38    rectangle
39}

41fn track_color(track: usize) -> AlphaColor<Srgb> {
42    const COLORS: [AlphaColor<Srgb>; 11] = [
43        manim::BLUE_C,
44        manim::TEAL_C,
45        manim::GREEN_C,
46        manim::YELLOW_C,
47        manim::GOLD_C,
48        manim::ORANGE,
49        manim::RED_C,
50        manim::MAROON_C,
51        manim::PURPLE_C,
52        manim::PINK,
53        manim::LIGHT_PINK,
54    ];
55    COLORS[track % COLORS.len()]
56}

58fn smoothstep(alpha: f64) -> f64 {
59    let alpha = alpha.clamp(0.0, 1.0);
60    alpha * alpha * (3.0 - 2.0 * alpha)
61}

63#[derive(Clone, Copy)]
64pub(crate) struct PianoLayout {
65    min_key: u8,
66    max_key: u8,
67    pitch_step: f64,
68    note_height: f64,
69    center_key: f64,
70}
71
72impl PianoLayout {
73    pub(crate) fn new(song: &MidiSong) -> Self {
74        let min_key = song.notes.iter().map(|note| note.key).min().unwrap_or(48);
75        let max_key = song.notes.iter().map(|note| note.key).max().unwrap_or(84);
76        let pitch_count = u16::from(max_key) - u16::from(min_key) + 1;
77        let pitch_step = (FRAME_HEIGHT - 0.5) / f64::from(pitch_count);
78        Self {
79            min_key,
80            max_key,
81            pitch_step,
82            note_height: pitch_step * 0.76,
83            center_key: (f64::from(min_key) + f64::from(max_key)) * 0.5,
84        }
85    }
86
87    fn y(self, key: u8) -> f64 {
88        (f64::from(key) - self.center_key) * self.pitch_step
89    }
90}
91
92fn is_black_key(key: u8) -> bool {
93    matches!(key % 12, 1 | 3 | 6 | 8 | 10)
94}

96pub(crate) fn make_background(layout: PianoLayout) -> Vec<VItem> {
97    let roll_width = FRAME_RIGHT - HIT_X;
98    let roll_center = (HIT_X + FRAME_RIGHT) * 0.5;
99    let mut background = vec![rectangle(
100        roll_width,
101        FRAME_HEIGHT,
102        dvec3(roll_center, 0.0, -0.05),
103        manim::GREY_E,
104        0.08,
105    )];
106
107    for key in layout.min_key..=layout.max_key {
108        let y = layout.y(key);
109        if is_black_key(key) {
110            background.push(rectangle(
111                roll_width,
112                layout.pitch_step,
113                dvec3(roll_center, y, -0.04),
114                manim::GREY_D,
115                0.14,
116            ));
117        }
118        if key % 12 == 0 {
119            background.push(rectangle(
120                roll_width,
121                0.018,
122                dvec3(roll_center, y - layout.pitch_step * 0.5, -0.03),
123                manim::GREY_B,
124                0.42,
125            ));
126        }
127    }
128
129    background.push(rectangle(
130        0.045,
131        FRAME_HEIGHT,
132        dvec3(HIT_X, 0.0, 0.15),
133        manim::WHITE,
134        0.9,
135    ));
136    background
137}

139fn render_note(note: Note, sec: f64, layout: PianoLayout, items: &mut Vec<VItem>) {
140    if note.end_sec <= sec {
141        return;
142    }
143
144    let raw_left = HIT_X + (note.start_sec - sec) * NOTE_SPEED;
145    let raw_right = HIT_X + (note.end_sec - sec) * NOTE_SPEED;
146    if raw_right <= HIT_X || raw_left >= FRAME_RIGHT {
147        return;
148    }
149
150    let left = raw_left.max(HIT_X);
151    let right = raw_right.min(FRAME_RIGHT + 0.1);
152    let width = right - left;
153    if width <= 0.002 {
154        return;
155    }
156
157    let hit_age = sec - note.start_sec;
158    let hit_pulse = if (0.0..HIT_EFFECT_SECS).contains(&hit_age) {
159        1.0 - smoothstep(hit_age / HIT_EFFECT_SECS)
160    } else {
161        0.0
162    };
163    let velocity = f64::from(note.velocity) / 127.0;
164    let height = layout.note_height * (1.0 + 0.28 * hit_pulse * velocity);
165    let opacity = (0.48 + 0.46 * velocity + 0.06 * hit_pulse) as f32;
166    let y = layout.y(note.key);
167    let z = 0.2 + note.track as f64 * 0.0002;
168
169    items.push(rectangle(
170        width,
171        height,
172        dvec3((left + right) * 0.5, y, z),
173        track_color(note.track),
174        opacity,
175    ));
176
177    if hit_pulse > 0.0 {
178        items.push(rectangle(
179            0.035 + 0.08 * hit_pulse,
180            height * (1.0 + 0.5 * hit_pulse),
181            dvec3(HIT_X + 0.025, y, z + 0.02),
182            manim::WHITE,
183            (0.2 + 0.7 * hit_pulse * velocity) as f32,
184        ));
185    }
186}

188pub(crate) struct MidiNotesEval {
189    pub(crate) song: Arc<MidiSong>,
190    pub(crate) layout: PianoLayout,
191}
192
193impl Eval for MidiNotesEval {
194    type Output = Vec<VItem>;
195
196    fn eval_alpha(&self, alpha: f64) -> Self::Output {
197        let sec = alpha.clamp(0.0, 1.0) * self.song.duration_secs;
198        let visible_until = sec + (FRAME_RIGHT - HIT_X) / NOTE_SPEED;
199        let first_index = self
200            .song
201            .notes
202            .partition_point(|note| note.start_sec < sec - self.song.max_note_duration_secs);
203        let end_index = self
204            .song
205            .notes
206            .partition_point(|note| note.start_sec <= visible_until);
207        let mut items = Vec::new();
208
209        for &note in &self.song.notes[first_index..end_index] {
210            render_note(note, sec, self.layout, &mut items);
211        }
212        items
213    }
214}

216pub(crate) struct SingleNoteEval {
217    note: Note,
218    layout: PianoLayout,
219    start_sec: f64,
220    duration_secs: f64,
221}
222
223impl SingleNoteEval {
224    pub(crate) fn new(note: Note, layout: PianoLayout) -> Self {
225        let visible_secs = (FRAME_RIGHT - HIT_X) / NOTE_SPEED;
226        let start_sec = (note.start_sec - visible_secs).max(0.0);
227        Self {
228            note,
229            layout,
230            start_sec,
231            duration_secs: note.end_sec - start_sec,
232        }
233    }
234
235    pub(crate) fn start_sec(&self) -> f64 {
236        self.start_sec
237    }
238
239    pub(crate) fn duration_secs(&self) -> f64 {
240        self.duration_secs
241    }
242}
243
244impl Eval for SingleNoteEval {
245    type Output = Vec<VItem>;
246
247    fn eval_alpha(&self, alpha: f64) -> Self::Output {
248        let sec = self.start_sec + alpha.clamp(0.0, 1.0) * self.duration_secs;
249        let mut items = Vec::with_capacity(2);
250        render_note(self.note, sec, self.layout, &mut items);
251        items
252    }
253}

255pub(crate) struct PianoKeyboardEval {
256    pub(crate) song: Arc<MidiSong>,
257    pub(crate) layout: PianoLayout,
258}
259
260impl PianoKeyboardEval {
261    fn key_state(&self, key: u8, sec: f64) -> (f64, usize) {
262        let mut strongest = 0.0_f64;
263        let mut strongest_track = 0;
264
265        for note in &self.song.notes_by_key[usize::from(key)] {
266            if note.start_sec > sec {
267                break;
268            }
269            if sec >= note.end_sec + KEY_RELEASE_SECS {
270                continue;
271            }
272
273            let envelope = if sec < note.start_sec + KEY_ATTACK_SECS {
274                smoothstep((sec - note.start_sec) / KEY_ATTACK_SECS)
275            } else if sec <= note.end_sec {
276                1.0
277            } else {
278                1.0 - smoothstep((sec - note.end_sec) / KEY_RELEASE_SECS)
279            };
280            let pressure = envelope * (0.5 + 0.5 * f64::from(note.velocity) / 127.0);
281            if pressure > strongest {
282                strongest = pressure;
283                strongest_track = note.track;
284            }
285        }
286
287        (strongest, strongest_track)
288    }
289}
290
291impl Eval for PianoKeyboardEval {
292    type Output = Vec<VItem>;
293
294    fn eval_alpha(&self, alpha: f64) -> Self::Output {
295        let sec = alpha.clamp(0.0, 1.0) * self.song.duration_secs;
296        let mut keys = Vec::new();
297
298        for key in self.layout.min_key..=self.layout.max_key {
299            let black = is_black_key(key);
300            let (pressure, track) = self.key_state(key, sec);
301            let width = if black {
302                KEYBOARD_WIDTH * 0.72
303            } else {
304                KEYBOARD_WIDTH
305            };
306            let pressed_offset = 0.095 * pressure;
307            let x = HIT_X - width * 0.5 - pressed_offset;
308            let y = self.layout.y(key);
309            let height = self.layout.pitch_step * if black { 0.78 } else { 0.92 };
310            let z = if black { 0.34 } else { 0.3 };
311            let base_color = if black { manim::GREY_E } else { manim::GREY_B };
312
313            keys.push(rectangle(width, height, dvec3(x, y, z), base_color, 0.96));
314
315            if pressure > 0.0 {
316                keys.push(rectangle(
317                    width,
318                    height,
319                    dvec3(x, y, z + 0.01),
320                    track_color(track),
321                    (0.18 + 0.72 * pressure) as f32,
322                ));
323                keys.push(rectangle(
324                    0.065,
325                    height * 0.86,
326                    dvec3(HIT_X - pressed_offset - 0.04, y, z + 0.02),
327                    manim::WHITE,
328                    (0.2 + 0.65 * pressure) as f32,
329                ));
330            }
331        }
332
333        keys
334    }
335}

337pub(crate) struct HitEffectsEval {
338    pub(crate) song: Arc<MidiSong>,
339    pub(crate) layout: PianoLayout,
340}
341
342impl Eval for HitEffectsEval {
343    type Output = Vec<VItem>;
344
345    fn eval_alpha(&self, alpha: f64) -> Self::Output {
346        let sec = alpha.clamp(0.0, 1.0) * self.song.duration_secs;
347        let first = self
348            .song
349            .notes
350            .partition_point(|note| note.start_sec < sec - HIT_EFFECT_SECS);
351        let last = self
352            .song
353            .notes
354            .partition_point(|note| note.start_sec <= sec);
355        let mut effects = Vec::new();
356
357        for note in &self.song.notes[first..last] {
358            let progress = ((sec - note.start_sec) / HIT_EFFECT_SECS).clamp(0.0, 1.0);
359            let opacity = 1.0 - smoothstep(progress);
360            let velocity = f64::from(note.velocity) / 127.0;
361            effects.push(rectangle(
362                0.06 + 0.34 * progress,
363                self.layout.note_height * (1.2 + 3.5 * progress),
364                dvec3(HIT_X + 0.04 * progress, self.layout.y(note.key), 0.42),
365                track_color(note.track),
366                (opacity * velocity * 0.62) as f32,
367            ));
368        }
369
370        effects
371    }
