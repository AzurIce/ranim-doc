14#[scene]
15#[wasm_demo_doc]
16#[output(dir = "./output/hello_ranim")]
17fn hello_ranim(r: &mut RanimScene) {
18    let mut square = Square::new(2.0);
19    square.set_color(manim::BLUE_C);
20
21    let mut content = seq![square.clone().fade_in().with_rate_func(smooth)];
22
23    let mut circle: VItem = Circle::new(2.0)
24        .with(|circle| {
25            circle.set_color(manim::RED_C);
26        })
27        .into();
28    circle.rotate_on_z(PI / 4.0 - PI);
29
30    let mut vitem = VItem::from(square);
31
32    content.extend(seq![
33        vitem.morph_to(circle).with_rate_func(smooth),
34        vitem.show(),
35        vitem.clone().unwrite().with_rate_func(smooth),
36        vitem.write().with_rate_func(smooth),
37        vitem.fade_out().with_rate_func(smooth),
38    ]);
39    r.play(
40        CameraFrame::default()
41            .show()
42            .with_duration(content.cursor_sec()),
43    );
44    r.play(content);
45
46    r.insert_time_mark(3.7, TimeMark::Capture("preview.png".to_string()));
47}
