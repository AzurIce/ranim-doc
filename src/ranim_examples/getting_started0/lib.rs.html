1use ranim::{
2    anims::fading::FadingAnim, color::palettes::manim, items::vitem::geometry::Square, prelude::*,
3    utils::rate_functions::smooth,

5
6// ANCHOR: construct
7#[scene]
8#[wasm_demo_doc]
9#[output(dir = "./output/getting_started0")]
10fn getting_started0(r: &mut RanimScene) {
11    // A Square with size 2.0 and color blue
12    let square = Square::new(2.0).with(|square| {
13        square.set_color(manim::BLUE_C);
14    });
15
16    let mut content = seq![square.clone().fade_in().with_rate_func(smooth)];
17    content
18        .hold(1.0)
19        .push(square.hide())
20        .forward(1.0)
21        .push(square.show())
22        .hold(1.0)
23        .push(square.clone().fade_out().with_rate_func(smooth));
24
25    r.play(
26        CameraFrame::default()
27            .show()
28            .with_duration(content.cursor_sec()),
29    );
30    r.play(content);
31    r.insert_time_mark(1.0, TimeMark::Capture("preview.png".to_string()));
32}
