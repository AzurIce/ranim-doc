17const TOTAL_SECS: f64 = INTRO_SECS + ORBIT_SECS;

19/// Something that can be positioned on an orbit.
20///
21/// The same orbital evaluator can therefore animate both mesh surfaces and
22/// vector-item groups without erasing either output type.
23trait OrbitItem: Clone {
24    fn set_orbit_position(&mut self, position: DVec3);
25}
26
27impl OrbitItem for Transformed<Surface, Translation> {
28    fn set_orbit_position(&mut self, position: DVec3) {
29        self.transform = Translation(position);
30    }
31}
32
33impl OrbitItem for Vec<VItem> {
34    fn set_orbit_position(&mut self, position: DVec3) {
35        self.move_to(position);
36    }
37}

39/// A reusable orbital evaluator whose output keeps its concrete item type.
40struct OrbitMotion<T> {
41    src: T,
42    orbit_radius: f64,
43    initial_angle: f64,
44    total_angle: f64,
45    z: f64,
46}
47
48impl<T: OrbitItem> Eval for OrbitMotion<T> {
49    type Output = T;
50
51    fn eval_alpha(&self, alpha: f64) -> Self::Output {
52        let angle = self.initial_angle + self.total_angle * alpha;
53        let position = DVec3::new(
54            self.orbit_radius * angle.cos(),
55            self.orbit_radius * angle.sin(),
56            self.z,
57        );
58        let mut result = self.src.clone();
59        result.set_orbit_position(position);
60        result
61    }
62}

131/// Build the typed orbital evaluator reused by every planet layer and label.
132fn orbit<T: OrbitItem>(item: &mut T, planet: &PlanetData, z: f64) -> OrbitMotion<T> {
133    OrbitMotion {
134        src: item.clone(),
135        orbit_radius: planet.orbit_radius,
136        initial_angle: planet.initial_angle,
137        total_angle: TAU * (ORBIT_SECS / planet.period),
138        z,
139    }
140    .apply_to(item)
141}

143/// One body layer: stay visible through the intro, then enter the shared orbit.
144fn body_layer(mut surface: Transformed<Surface, Translation>, planet: &PlanetData) -> AnimSequence {
145    seq![
146        surface.show().with_duration(INTRO_SECS),
147        orbit(&mut surface, planet, 0.0)
148            .with_duration(ORBIT_SECS)
149            .with_rate_func(linear),
150    ]
151}

153/// A label has its own intro phrase, followed by the same reusable orbit phase.
154fn planet_label(planet: &PlanetData, position: DVec3) -> AnimSequence {
155    let label_z = planet.radius + 0.5;
156    let mut label = Vec::<VItem>::from(TextItem::new(planet.name, 0.6));
157    label
158        .move_to(position.with_z(label_z))
159        .with_origin(AabbPoint::CENTER, |items| {
160            items.rotate_on_x(30.0f64.to_radians()).discard()
161        });
162
163    let mut sequence = AnimSequence::new();
164    sequence
165        .forward(LABEL_DELAY_SECS)
166        .push(
167            label
168                .fade_in()
169                .with_duration(LABEL_FADE_SECS)
170                .with_rate_func(smooth),
171        )
172        .push(
173            orbit(&mut label, planet, label_z)
174                .with_duration(ORBIT_SECS)
175                .with_rate_func(linear),
176        );
177    sequence
178}

180/// Build a planet as a nested stack of body layers and its label track.
181///
182/// Atmospheric planets reuse the same body-layer phrase twice; rocky planets
183/// use it once. The outer stack keeps each planet as one semantic Timeline node.
184fn planet_system(planet: &PlanetData) -> AnimStack {
185    let position = DVec3::new(
186        planet.orbit_radius * planet.initial_angle.cos(),
187        planet.orbit_radius * planet.initial_angle.sin(),
188        0.0,
189    );
190    let mut body_layers = AnimStack::new();
191
192    if planet.has_atmosphere {
193        let core = Surface::from(
194            Sphere::new(planet.radius * 0.9)
195                .with_resolution((20, 10))
196                .with_fill_color(planet.color.with_alpha(1.0)),
197        )
198        .transformed(Translation(position));
199        body_layers.push(body_layer(core, planet));
200
201        let atmosphere = Surface::from(
202            Sphere::new(planet.radius)
203                .with_resolution((20, 10))
204                .with_fill_color(planet.color.with_alpha(0.3)),
205        )
206        .with_smooth_normals()
207        .transformed(Translation(position));
208        body_layers.push(body_layer(atmosphere, planet));
209    } else {
210        let surface = Surface::from(
211            Sphere::new(planet.radius)
212                .with_resolution((20, 10))
213                .with_fill_color(planet.color.with_alpha(1.0)),
214        )
215        .with_smooth_normals()
216        .transformed(Translation(position));
217        body_layers.push(body_layer(surface, planet));
218    }
219
220    stack![body_layers, planet_label(planet, position)]
221}

223/// Fade one orbit ring in, then hold its final state for the full scene.
224fn orbit_ring(planet: &PlanetData) -> AnimSequence {
225    let major_radius = planet.orbit_radius;
226    let minor_radius = 0.05;
227    let mut ring = Surface::from_uv_func(
228        move |u, v| {
229            let u_angle = u * TAU;
230            let v_angle = v * TAU;
231            DVec3::new(
232                (major_radius + minor_radius * v_angle.cos()) * u_angle.cos(),
233                (major_radius + minor_radius * v_angle.cos()) * u_angle.sin(),
234                minor_radius * v_angle.sin(),
235            )
236        },
237        (0.0, 1.0),
238        (0.0, 1.0),
239        (128, 16),
240    );
241    let color = manim::GREY_C.with_alpha(0.3);
242    ring.vertex_colors = vec![color; ring.vertices.len()];
243
244    let mut sequence = seq![ring.fade_in().with_duration(1.0).with_rate_func(smooth),];
245    sequence.hold_to(TOTAL_SECS);
246    sequence
247}
248
249fn orbit_rings() -> AnimStack {
250    let mut rings = AnimStack::new();
251    for planet in PLANETS {
252        rings.push(orbit_ring(planet));
253    }
254    rings
255}

257fn planetary_systems() -> AnimStack {
258    let mut planets = AnimStack::new();
259    for planet in PLANETS {
260        planets.push(planet_system(planet));
261    }
262    planets
263}

265#[scene]
266#[wasm_demo_doc]
267#[output(dir = "./output/solar_system")]
268fn solar_system(r: &mut RanimScene) {
269    let phi = 50.0f64.to_radians();
270    let theta = -PI / 2.0;
271    let distance = 60.0;
272    let mut camera = CameraFrame::from_spherical(phi, theta, distance);
273    camera.fovy = 30.0f64.to_radians();
274
275    let sun = Surface::from(
276        Sphere::new(2.0)
277            .with_resolution((30, 15))
278            .with_fill_color(manim::YELLOW_C.with_alpha(0.5)),
279    );
280    // Stack -> Stack -> Sequence: orbit rings and reusable planet systems
281    // remain visible as semantic groups in the preview Timeline.
282    r.play(camera.show().with_duration(TOTAL_SECS));
283    r.play(stack![
284        sun.show().with_duration(TOTAL_SECS),
285        orbit_rings(),
286        planetary_systems(),
287    ]);
288    r.insert_time_mark(6.0, TimeMark::Capture("preview.png".to_owned()));
289}
