19/// Extract an evaluated value into render-world entities or resources.
20///
21/// Implementations can recurse into other values using the same context. Each
22/// call to [`ExtractContext::entity`] claims the next persistent output of the
23/// source frame item. Outputs no longer claimed on the next extraction are
24/// despawned automatically. Entities created directly through
25/// [`ExtractContext::world_mut`] remain the implementation's responsibility.
26pub trait Extract {
27    /// Synchronize this value's render-world outputs.
28    fn extract(&self, ctx: &mut ExtractContext<'_>);
29}
30
31/// Installs extraction for one exact evaluated frame payload type.
32///
33/// For example, a frame containing `Vec<T>` needs `ExtractPlugin<Vec<T>>`;
34/// installing `ExtractPlugin<T>` alone does not visit that batch. Recursive
35/// [`Extract`] implementations handle the values inside the registered type.
36/// Handwritten systems can also be added to [`RenderSchedule::Extract`].
37pub struct ExtractPlugin<T>(PhantomData<fn() -> T>);
38
39impl<T> Default for ExtractPlugin<T> {
40    fn default() -> Self {
41        Self(PhantomData)
42    }
43}
44
45impl<T: Extract + Send + Sync + 'static> Plugin for ExtractPlugin<T> {
46    fn build(&self, app: &mut App) {
47        app.init_resource::<ExtractedEntities<T>>()
48            .add_systems(RenderSchedule::Extract, extract_items::<T>);
49    }
50}

54struct OutputEntity {

59#[derive(Resource)]
60struct ExtractedEntities<T: Send + Sync + 'static> {
61    entities: HashMap<OutputId, OutputEntity>,
62    seen: HashSet<OutputId>,
63    inputs: HashSet<FrameItemId>,
64    marker: PhantomData<fn() -> T>,
65}
66
67impl<T: Send + Sync + 'static> Default for ExtractedEntities<T> {
68    fn default() -> Self {
69        Self {
70            entities: HashMap::new(),
71            seen: HashSet::new(),
72            inputs: HashSet::new(),
73            marker: PhantomData,
74        }
75    }

78/// World access, output identity, order, and inherited transform for extraction.
79///
80/// A context belongs to one [`ranim_core::frame::FrameItem`]. Recursive values
81/// share its output sequence; transform scopes compose without changing that
82/// sequence. Input data remains unchanged when a frame is rendered again.
83pub struct ExtractContext<'w> {
84    world: &'w mut World,
85    entities: &'w mut HashMap<OutputId, OutputEntity>,
86    seen: &'w mut HashSet<OutputId>,
87    source: FrameItemId,
88    order: usize,
89    next_part: &'w mut usize,
90    transform: DAffine3,
91}
92
93impl ExtractContext<'_> {
94    /// Access the render world to update resources or perform custom operations.
95    ///
96    /// The current frame and this plugin's identity state are temporarily scoped
97    /// out of the world while extracting. The input value is available as `self`
98    /// in [`Extract::extract`]. Directly spawned entities are not tracked by the
99    /// context; manage their lifetimes in the implementation.
100    pub fn world_mut(&mut self) -> &mut World {
101        self.world
102    }
103
104    /// Claim the source's next persistent output entity.
105    ///
106    /// `B` identifies the output bundle's shape. The entity is reused while this
107    /// shape stays the same; changing it replaces the entity so old components
108    /// cannot remain. Values are inserted by the caller. Components managed
109    /// outside `B` remain the implementation's responsibility.
110    /// An output not claimed on a later frame is despawned.
111    pub fn entity<B: Bundle>(&mut self) -> EntityWorldMut<'_> {
112        self.next_entity(TypeId::of::<B>())
113    }
114
115    /// The accumulated local-to-world transform for the current value.
116    pub fn transform(&self) -> DAffine3 {
117        self.transform
118    }
119
120    /// Extract a subtree under `current * transform`, sharing output identities.
121    pub fn with_transform<R>(
122        &mut self,
123        transform: DAffine3,
124        extract: impl FnOnce(&mut ExtractContext<'_>) -> R,
125    ) -> R {
126        let mut child = ExtractContext {
127            world: self.world,
128            entities: self.entities,
129            seen: self.seen,
130            source: self.source,
131            order: self.order,
132            next_part: self.next_part,
133            transform: self.transform * transform,
134        };
135        extract(&mut child)
136    }
137
138    fn next_entity(&mut self, bundle_type: TypeId) -> EntityWorldMut<'_> {
139        let part = *self.next_part;
140        *self.next_part += 1;
141        let id = (self.source, part);
142        self.seen.insert(id);
143
144        let output = self.entities.entry(id).or_insert_with(|| OutputEntity {
145            entity: self.world.spawn_empty().id(),
146            bundle_type,
147        });
148        if output.bundle_type != bundle_type || !self.world.entities().contains(output.entity) {
149            self.world.despawn(output.entity);
150            output.entity = self.world.spawn_empty().id();
151            output.bundle_type = bundle_type;
152        }
153
154        let mut entity = self.world.entity_mut(output.entity);
155        let order = SourceOrder {
156            order: self.order,
157            part,
158        };
159        if entity.get::<SourceOrder>() != Some(&order) {
160            entity.insert(order);
161        }
162        entity
163    }
164}

166fn extract_items<T: Extract + Send + Sync + 'static>(world: &mut World) {
167    let _span = crate::cpu_probe::span("reconcile");
168    world.resource_scope(|world, mut state: Mut<ExtractedEntities<T>>| {
169        world.resource_scope(|world, frame: Mut<EvaluatedFrame>| {
170            let state = &mut *state;
171            state.seen.clear();
172            state.inputs.clear();
173            for item in frame.items::<T>() {
174                assert!(
175                    state.inputs.insert(item.id),
176                    "duplicate frame item id {:?} for `{}`",
177                    item.id,
178                    std::any::type_name::<T>()
179                );
180                let mut next_part = 0;
181                item.value.extract(&mut ExtractContext {
182                    world,
183                    entities: &mut state.entities,
184                    seen: &mut state.seen,
185                    source: item.id,
186                    order: item.order,
187                    next_part: &mut next_part,
188                    transform: DAffine3::IDENTITY,
189                });
190            }
191            state.entities.retain(|id, output| {
192                if state.seen.contains(id) {
193                    true
194                } else {
195                    world.despawn(output.entity);
196                    false
197                }
198            });
199        });
200    });
201}

203impl<T: Extract, G: Clone + Into<DAffine3>> Extract for Transformed<T, G> {
204    fn extract(&self, ctx: &mut ExtractContext<'_>) {
205        ctx.with_transform(self.transform.clone().into(), |ctx| self.inner.extract(ctx));
206    }
207}

209impl<T: Extract> Extract for Vec<T> {
210    fn extract(&self, ctx: &mut ExtractContext<'_>) {
211        for item in self {
212            item.extract(ctx);
213        }
214    }
215}

217impl<T: Extract> Extract for Option<T> {
218    fn extract(&self, ctx: &mut ExtractContext<'_>) {
219        if let Some(item) = self {
220            item.extract(ctx);
221        }
222    }
223}

225impl<T: Extract, const N: usize> Extract for [T; N] {
226    fn extract(&self, ctx: &mut ExtractContext<'_>) {
227        for item in self {
228            item.extract(ctx);
229        }
230    }
231}

237macro_rules! impl_extract_for_tuple {
238    ($(($T:ident, $value:ident)),*) => {
239        impl<$($T: Extract),*> Extract for ($($T,)*) {
240            fn extract(&self, ctx: &mut ExtractContext<'_>) {
241                let ($($value,)*) = self;
242                $($value.extract(ctx);)*
243            }
244        }
245    };
246}
