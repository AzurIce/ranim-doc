21/// Installs the built-in render plugins: camera frame, vector items, and mesh
22/// items. This is what the default render setup of a project binary uses;
23/// install it explicitly when composing a custom
24/// [`RenderSetup`](ranim_render::RenderSetup) with additional extraction
25/// plugins, and do not combine it with the individual plugins (they are
26/// unique in a bevy app).
27#[cfg(feature = "render")]
28pub struct RanimItemsPlugin;
29
30#[cfg(feature = "render")]
31impl bevy_app::Plugin for RanimItemsPlugin {
32    fn build(&self, app: &mut bevy_app::App) {
33        app.add_plugins((CameraFramePlugin, VItemPlugin, MeshItemPlugin));
34    }
